```xml
<?xml version="1.0" encoding="UTF-8"?>
<seir:SEIRModel xmi:version="2.0"
    xmlns:xmi="http://www.omg.org/XMI"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xmlns:seir="http://example.com/seirmodel">

  <!-- ===================== -->
  <!-- Parameters (symbolic) -->
  <!-- ===================== -->
  <parameters name="B" type="CONSTANT"
              description="Constant influx (birth/recruitment) rate into susceptible human hosts"
              unit="humans/time"/>
  <parameters name="μₕ" type="CONSTANT"
              description="Natural death rate of human hosts"
              unit="1/time"/>
  <parameters name="μᵥ" type="CONSTANT"
              description="Natural death rate of mosquitoes (vectors)"
              unit="1/time"/>

  <!-- Age-dependent (functions of age a) -->
  <parameters name="βₕ(a)" type="VARIABLE"
              description="Age-dependent transmission/contact rate from infectious mosquitoes to susceptible humans (vector-to-host)"
              unit="1/time"/>
  <parameters name="k(a)" type="VARIABLE"
              description="Age-dependent progression rate from exposed humans to infectious humans (end of latent period)"
              unit="1/time"/>
  <parameters name="γ₁(a)" type="VARIABLE"
              description="Age-dependent recovery rate of symptomatic infectious humans"
              unit="1/time"/>
  <parameters name="γ₂(a)" type="VARIABLE"
              description="Age-dependent recovery rate of asymptomatic infectious humans"
              unit="1/time"/>

  <!-- Host-to-vector transmission (age-dependent per text Table list) -->
  <parameters name="βᵥ(a)" type="VARIABLE"
              description="Age-dependent transmission rate from infectious human hosts to susceptible mosquitoes (host-to-vector)"
              unit="1/time"/>

  <!-- Fractions / modifiers -->
  <parameters name="ρ" type="CONSTANT"
              description="Probability an exposed human becomes symptomatic infectious (fraction progressing to Iₕ); (1-ρ) becomes asymptomatic Aₕ"
              unit="dimensionless"/>
  <parameters name="σ" type="CONSTANT"
              description="Reinfection intensity factor: scales contact rate for reinfection of asymptomatic humans leading to symptomatic infection"
              unit="dimensionless"/>

  <!-- Normalization term used in incidence (kept symbolic) -->
  <parameters name="Nₕ(t)" type="VARIABLE"
              description="Total human population size at time t (used in standard incidence normalization)"
              unit="humans"/>

  <!-- ===================== -->
  <!-- Compartments (states) -->
  <!-- ===================== -->
  <!-- Humans are age-structured densities: X_h(a,t). Mosquitoes are time-only: X_v(t). -->
  <compartments PrimaryName="Susceptible" SecondaryName="Human (age-structured)" population="0.0">
    <!-- Infection by infectious mosquitoes (standard incidence shown in paper/figure) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.3"
                   contactCompartment="//@compartments.5"
                   target="//@compartments.1"
                   description="Vector-to-host infection: Sₕ(a,t) -> Eₕ(a,t) at rate βₕ(a) * Iᵥ(t) / Nₕ(t) (standard incidence; Nₕ(t) handled externally)"/>
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="Human (age-structured)" population="0.0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.4"
                   target="//@compartments.2"
                   description="Progression to symptomatic infectious: Eₕ(a,t) -> Iₕ(a,t) at rate ρ * k(a) (ρ applied externally as fraction)"/>
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.4"
                   target="//@compartments.3"
                   description="Progression to asymptomatic infectious: Eₕ(a,t) -> Aₕ(a,t) at rate (1-ρ) * k(a) ((1-ρ) applied externally as fraction)"/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Human Symptomatic (age-structured)" population="0.0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.5"
                   target="//@compartments.4"
                   description="Recovery of symptomatic humans: Iₕ(a,t) -> Rₕ(a,t) at rate γ₁(a)"/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Human Asymptomatic (age-structured)" population="0.0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.6"
                   target="//@compartments.4"
                   description="Recovery of asymptomatic humans: Aₕ(a,t) -> Rₕ(a,t) at rate γ₂(a)"/>
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.3"
                   contactCompartment="//@compartments.5"
                   target="//@compartments.2"
                   description="Reinfection/secondary infection: Aₕ(a,t) -> Iₕ(a,t) at rate σ * βₕ(a) * Iᵥ(t) / Nₕ(t) (σ applied externally as multiplier)"/>
  </compartments>

  <compartments PrimaryName="Recovered" SecondaryName="Human (age-structured)" population="0.0">
    <!-- No outgoing flows (decoupled in paper) -->
  </compartments>

  <compartments PrimaryName="Susceptible" SecondaryName="Mosquito (vector)" population="0.0">
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.7"
                   contactCompartment="//@compartments.2"
                   target="//@compartments.6"
                   description="Host-to-vector infection driven by infectious humans (symptomatic and asymptomatic contribute in full model): Sᵥ(t) -> Iᵥ(t) at rate βᵥ(a) * (Iₕ + Aₕ) (age-aggregation handled externally)"/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Mosquito (vector)" population="0.0">
    <!-- No recovery for vectors -->
  </compartments>

  <!-- ================= -->
  <!-- Birth / recruitment -->
  <!-- ================= -->
  <birthSources name="HumanRecruitment"
                rateParameter="//@parameters.0"
                targetCompartment="//@compartments.0"
                fixedRate="true"/>

  <!-- ================= -->
  <!-- Death sinks -->
  <!-- ================= -->
  <!-- Human natural deaths from all human compartments -->
  <deathSinks name="NaturalDeath_S_h" rateParameter="//@parameters.1" sourceCompartment="//@compartments.0"/>
  <deathSinks name="NaturalDeath_E_h" rateParameter="//@parameters.1" sourceCompartment="//@compartments.1"/>
  <deathSinks name="NaturalDeath_I_h" rateParameter="//@parameters.1" sourceCompartment="//@compartments.2"/>
  <deathSinks name="NaturalDeath_A_h" rateParameter="//@parameters.1" sourceCompartment="//@compartments.3"/>
  <deathSinks name="NaturalDeath_R_h" rateParameter="//@parameters.1" sourceCompartment="//@compartments.4"/>

  <!-- Mosquito natural deaths from both vector compartments -->
  <deathSinks name="NaturalDeath_S_v" rateParameter="//@parameters.2" sourceCompartment="//@compartments.5"/>
  <deathSinks name="NaturalDeath_I_v" rateParameter="//@parameters.2" sourceCompartment="//@compartments.6"/>

</seir:SEIRModel>
```
