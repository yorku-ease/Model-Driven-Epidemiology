<?xml version="1.0" encoding="UTF-8"?>
<seir:SEIRModel xmi:version="2.0"
    xmlns:xmi="http://www.omg.org/XMI"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xmlns:seir="http://example.com/seirmodel">

  <!-- ===================== -->
  <!-- Parameters (symbolic) -->
  <!-- ===================== -->
  <parameters name="A" type="CONSTANT"
              description="Recruitment (birth/immigration) rate into the susceptible population"
              unit="persons/time"/>
  <parameters name="μ" type="CONSTANT"
              description="Natural per-capita death rate (applies to all compartments)"
              unit="1/time"/>
  <parameters name="u" type="CONSTANT"
              description="Vaccination rate of susceptible individuals (S→V)"
              unit="1/time"/>
  <parameters name="θ" type="CONSTANT"
              description="Waning rate of vaccine-induced protection (V→S)"
              unit="1/time"/>
  <parameters name="β" type="CONSTANT"
              description="Transmission coefficient (mass-action incidence) driving infection of susceptibles (S→I) via contact with infected"
              unit="1/(persons·time)"/>
  <parameters name="ε" type="CONSTANT"
              description="Relative susceptibility (or reinfection modifier) for recovered individuals; reinfection term uses εβ"
              unit="dimensionless"/>
  <parameters name="γ" type="CONSTANT"
              description="Recovery rate from infection (I→R)"
              unit="1/time"/>
  <parameters name="δ" type="CONSTANT"
              description="Waning rate of natural immunity after recovery (R→S)"
              unit="1/time"/>

  <!-- ===================== -->
  <!-- Compartments (states) -->
  <!-- ===================== -->
  <compartments PrimaryName="Susceptible" population="0">
    <!-- Infection: S -> I (mass-action with I) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRate="0.0"
                   contactRateParameter="//@parameters.5"
                   contactCompartment="//@compartments.2"
                   target="//@compartments.2"
                   description="Primary infection of susceptibles at rate β*S*I"/>
    <!-- Vaccination: S -> V -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rate="0.0"
                   rateParameter="//@parameters.2"
                   target="//@compartments.1"
                   description="Vaccination of susceptibles at rate u*S"/>
  </compartments>

  <compartments PrimaryName="Vaccinated" population="0">
    <!-- Waning vaccine immunity: V -> S -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rate="0.0"
                   rateParameter="//@parameters.3"
                   target="//@compartments.0"
                   description="Waning of vaccine-induced protection at rate θ*V"/>
  </compartments>

  <compartments PrimaryName="Infected" SecondaryName="Infectious" population="0">
    <!-- Recovery: I -> R -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rate="0.0"
                   rateParameter="//@parameters.7"
                   target="//@compartments.3"
                   description="Recovery from infection at rate γ*I"/>
  </compartments>

  <compartments PrimaryName="Recovered" population="0">
    <!-- Loss of natural immunity: R -> S -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rate="0.0"
                   rateParameter="//@parameters.8"
                   target="//@compartments.0"
                   description="Waning of natural immunity at rate δ*R"/>
    <!-- Reinfection: R -> I (mass-action with I, scaled by ε) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRate="0.0"
                   contactRateParameter="//@parameters.5"
                   contactCompartment="//@compartments.2"
                   target="//@compartments.2"
                   description="Reinfection of recovered at rate ε*β*R*I (ε is a modifier parameter)"/>
  </compartments>

  <!-- ============================== -->
  <!-- Demography: births and deaths  -->
  <!-- ============================== -->

  <!-- Recruitment into Susceptible: A -> S -->
  <birthSources name="Recruitment"
                rate="0.0"
                rateParameter="//@parameters.0"
                targetCompartment="//@compartments.0"/>

  <!-- Natural deaths from each compartment: μ * (compartment) -->
  <deathSinks name="Natural_Death_S"
              rate="0.0"
              rateParameter="//@parameters.1"
              sourceCompartment="//@compartments.0"/>
  <deathSinks name="Natural_Death_V"
              rate="0.0"
              rateParameter="//@parameters.1"
              sourceCompartment="//@compartments.1"/>
  <deathSinks name="Natural_Death_I"
              rate="0.0"
              rateParameter="//@parameters.1"
              sourceCompartment="//@compartments.2"/>
  <deathSinks name="Natural_Death_R"
              rate="0.0"
              rateParameter="//@parameters.1"
              sourceCompartment="//@compartments.3"/>

</seir:SEIRModel>
