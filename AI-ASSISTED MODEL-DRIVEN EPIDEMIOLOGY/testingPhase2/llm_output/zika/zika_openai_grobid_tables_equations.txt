  <parameters name="e_V" type="CONSTANT" expression="80"

  <parameters name="τ_1" type="CONSTANT" expression="0.3333333333"

              unit="per day"/>
  <parameters name="μ_3" type="CONSTANT" expression="0.0025"
              description="Pupal mortality rate"
              unit="per day"/>
  <parameters name="μ_V" type="CONSTANT" expression="0.05"
              description="Adult female mosquito mortality rate"
              unit="per day"/>

  <parameters name="q" type="CONSTANT" expression="0.68"

  <parameters name="K_V" type="CONSTANT" expression="100000000"

  <parameters name="κ" type="CONSTANT" expression="1"

              unit="per day"/>

  <parameters name="α_V" type="CONSTANT" expression="0.1"

              unit="per day"/>

         - Logistic crowding loss term: (λ*N_H/K_H)*S_H
       Inflow:
         - Human logistic recruitment term: λ*N_H*(1 - S_H/K_H) (implemented as BirthSource below; crowding part remains as outflow)
    -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.2"
                   contactCompartment="//@compartments.9"
                   target="//@compartments.1"
                   description="Human infection from bites by infectious adult female mosquitoes (β*b*I_V/(N_H+M))*S_H; denominator handled externally via parameterization using N_H and M."/>
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.0"
                   target="//@compartments.10"

    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.0"
                   target="//@compartments.10"

  </compartments>

  <!-- 2 -->
  <compartments PrimaryName="Infectious" SecondaryName="Humans" population="0">

    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.0"
                   target="//@compartments.10"
                   description="Logistic crowding loss from infectious humans: (λ*N_H/K_H)*I_H (modeled as removal to sink compartment)."/>
  </compartments>

  <!-- 3 -->
  <compartments PrimaryName="Recovered" SecondaryName="Humans" population="0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.0"

  </compartments>

  <!-- 4 -->
  <compartments PrimaryName="Eggs" SecondaryName="Vectors" population="0">

    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.11"
                   target="//@compartments.11"
                   description="Egg mortality: μ_1 * V_1 (modeled as removal to sink compartment 'Vector deaths')."/>

  <!-- 5 -->
  <compartments PrimaryName="Larvae" SecondaryName="Vectors" population="0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.9"
                   target="//@compartments.6"

    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.12"
                   target="//@compartments.11"

    <!-- Density-dependent larval mortality term: κ * V2^2 / K_V (nonlinear; represented as EXPRESSION parameter is not possible here without custom flow type).

    <outgoingFlows xsi:type="seir:RateFlow"
                   rate="0.0"
                   target="//@compartments.11"
                   description="Density-dependent larval mortality: κ * V_2^2 / K_V (nonlinear term; must be implemented in simulation layer)."/>
  </compartments>

  <!-- 6 -->
  <compartments PrimaryName="Pupae" SecondaryName="Vectors" population="0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.10"
                   target="//@compartments.7"
                   description="Pupal development to adult females (before infection status split): τ_3 * V_3 contributes to adult recruitment (q*τ_3*V_3 to susceptible adult females)."/>
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.13"
                   target="//@compartments.11"
                   description="Pupal mortality: μ_3 * V_3 (modeled as removal to sink compartment 'Vector deaths')."/>
  </compartments>

  <!-- 7 -->
  <compartments PrimaryName="Susceptible" SecondaryName="Adult female vectors" population="0">
    <outgoingFlows xsi:type="seir:ContactFlow"

                   target="//@compartments.8"
                   description="Vector infection from biting infectious humans: (β*b*I_H/(N_H+M))*S_V; denominator handled externally via N_H and M."/>
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.14"
                   target="//@compartments.11"
                   description="Adult female vector mortality from susceptible class: μ_V * S_V (to vector deaths sink)."/>
  </compartments>

  <!-- 8 -->
  <compartments PrimaryName="Exposed" SecondaryName="Adult female vectors" population="0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.18"
                   target="//@compartments.9"
                   description="Progression from exposed vectors to infectious vectors: α_V * E_V."/>
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.14"
                   target="//@compartments.11"
                   description="Adult female vector mortality from exposed class: μ_V * E_V (to vector deaths sink)."/>

  <!-- 9 -->
  <compartments PrimaryName="Infectious" SecondaryName="Adult female vectors" population="0">
    <outgoingFlows xsi:type="seir:RateFlow"

                   description="Adult female vector mortality from infectious class: μ_V * I_V (to vector deaths sink)."/>
  </compartments>

  <!-- 10: sink for logistic crowding loss terms (paper uses -λ N_H * X / K_H in each human compartment) -->
  <compartments PrimaryName="Removed" SecondaryName="Human logistic crowding loss" population="0"/>

  <compartments PrimaryName="Dead" SecondaryName="Vectors" population="0"/>

  <!-- Birth sources          -->
  <!-- ===================== -->

  <!-- Human logistic recruitment term: λ N_H (1 - S_H/K_H)
       Implemented as a BirthSource into Susceptible Humans.
       Note: exact dependence on N_H and S_H is nonlinear; simulation layer must compute it. -->
  <birthSources name="Human_Logistic_Recruitment"

                targetCompartment="//@compartments.0"

  <birthSources name="Adult_Female_Emergence"

                targetCompartment="//@compartments.7"
                fixedRate="false"/>

</seir:SEIRModel>
