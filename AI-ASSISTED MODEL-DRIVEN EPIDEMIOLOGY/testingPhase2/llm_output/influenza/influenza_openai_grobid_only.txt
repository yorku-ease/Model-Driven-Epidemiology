<?xml version="1.0" encoding="UTF-8"?>
<seir:SEIRModel xmi:version="2.0"
    xmlns:xmi="http://www.omg.org/XMI"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xmlns:seir="http://example.com/seirmodel">

  <!-- ===================== -->
  <!-- Parameters (symbolic) -->
  <!-- ===================== -->
  <parameters name="β" type="CONSTANT"
              description="Constant transmission rate (rate at which new infections take place given a contact); used in force of infection λ(t) proportional to infectious people."
              unit="per day"/>
  <parameters name="λ(t)" type="EXPRESSION"
              description="Force of infection (rate at which a susceptible becomes infected), assumed proportional to number of infectious people at time t and transmission rate β (homogeneous mixing)."
              unit="per day"/>

  <parameters name="σ" type="CONSTANT"
              description="Daily rate of transition E₁→E₂ and E₂→I₁; implies average incubation period d_L = 2*(1/σ) days."
              unit="per day"/>
  <parameters name="γ" type="CONSTANT"
              description="Daily rate of transition I₁→I₂ and I₂→R; implies average infectious period d_I = 2*(1/γ) days."
              unit="per day"/>

  <parameters name="π" type="CONSTANT"
              description="Initial susceptibility (proportion non-immune at start); used in effective reproduction number R_n together with epidemic growth rate parameter ψ."
              unit="dimensionless"/>

  <parameters name="λ₀" type="CONSTANT"
              description="Initial infection rate / initial force of infection parameter at epidemic start (as used in parameter set Θ_T = (π, λ₀, ψ, σ, γ))."
              unit="per day"/>

  <parameters name="ψ" type="CONSTANT"
              description="Epidemic growth-related parameter (EGR parametrisation); used with π to determine effective reproduction number R_n."
              unit="1/time"/>

  <parameters name="p_ICU" type="CONSTANT"
              description="Probability of ICU/HDU admission given infection."
              unit="dimensionless"/>

  <parameters name="μ_ICU|E" type="CONSTANT"
              description="Mean of Gamma-distributed waiting time from infection to ICU/HDU admission."
              unit="time"/>
  <parameters name="σ²_ICU|E" type="CONSTANT"
              description="Variance of Gamma-distributed waiting time from infection to ICU/HDU admission."
              unit="time^2"/>

  <parameters name="η" type="CONSTANT"
              description="Negative Binomial dispersion parameter for weekly ICU/HDU admission counts (observation model)."
              unit="dimensionless"/>

  <!-- Quantities mentioned but not fully specified numerically in provided sections -->
  <parameters name="R₀" type="EXPRESSION"
              description="Basic reproduction number; discussed with plausible range 1≤R₀≤4 used to set priors (no explicit formula given in provided sections)."
              unit="dimensionless"/>
  <parameters name="R_n" type="EXPRESSION"
              description="Effective reproduction number in partially immune population; stated to be a function of susceptibility π and EGR parameter ψ (explicit formula not provided in provided sections)."
              unit="dimensionless"/>
  <parameters name="TAR" type="EXPRESSION"
              description="Total Attack Rate: probability of being infected over the whole epidemic for a susceptible individual (computed from transmission model; explicit formula not provided in provided sections)."
              unit="dimensionless"/>

  <!-- ===================== -->
  <!-- Compartments (states) -->
  <!-- ===================== -->
  <compartments PrimaryName="Susceptible" population="0.0">
    <!-- Infection: S -> E1 via contact with infectious people (homogeneous mixing) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.0"
                   contactCompartment="//@compartments.3"
                   target="//@compartments.1"
                   description="Transmission: susceptible individuals become infected (enter latent stage E₁) at rate λ(t) proportional to β and infectious population."/>
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="E1" population="0.0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.2"
                   target="//@compartments.2"
                   description="Latent progression E₁→E₂ at daily rate σ."/>
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="E2" population="0.0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.2"
                   target="//@compartments.3"
                   description="Latent progression E₂→I₁ at daily rate σ."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="I1" population="0.0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.3"
                   target="//@compartments.4"
                   description="Infectious stage progression I₁→I₂ at daily rate γ."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="I2" population="0.0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.3"
                   target="//@compartments.5"
                   description="Recovery/removal I₂→R at daily rate γ."/>
  </compartments>

  <compartments PrimaryName="Recovered" population="0.0"/>

  <!-- Notes:
       - ICU/HDU admissions are not modeled as a compartment in the provided sections; they are an observation process:
         expected ICU admissions μ_t are a convolution of past infections with a Gamma waiting time, scaled by p_ICU,
         and weekly counts follow a Negative Binomial with dispersion η.
       - No births/deaths (BirthSource/DeathSink) are described in the provided sections.
       - Deterministic homogeneous-mixing SEIR with two-stage E and I is described; σ and γ are fixed from prior studies.
  -->

</seir:SEIRModel>
