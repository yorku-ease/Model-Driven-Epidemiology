<?xml version="1.0" encoding="UTF-8"?>
<seir:SEIRModel xmi:version="2.0"
    xmlns:xmi="http://www.omg.org/XMI"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xmlns:seir="http://example.com/seirmodel"
    totalPopulation="0">

  <!-- =========================================================
       Parameters (paper symbols; numeric values prioritized from Camelot tables)
       Notes:
       - δt is the discrete time step (quarter day) used in the paper's difference equations.
         The metamodel is continuous-rate oriented; δt is included as a CONSTANT parameter for documentation.
       - N_year values are included as parameters (season-specific), since the paper uses a fixed N per season.
       - R0, Rn, TAR, I0 are derived quantities in the paper; included as EXPRESSION placeholders (no single closed
         form fully specified in the provided text for all of them).
       ========================================================= -->
  <parameters name="β" type="CONSTANT"
      description="Transmission rate parameter in force of infection λ(t)=β I(t)/N (frequency-dependent)."
      unit="1/day"/>

  <parameters name="λ(t)" type="EXPRESSION"
      expression="β * I(t) / N"
      description="Force of infection at time t (per-susceptible infection hazard)."
      unit="1/day"/>

  <parameters name="δt" type="CONSTANT"
      expression="0.25"
      description="Discrete-time step used in the paper (quarter of a day)."
      unit="day"/>

  <parameters name="σ" type="CONSTANT"
      expression="1"
      description="Daily rate of progression through exposed stages (E1→E2 and E2→I1)."
      unit="1/day"/>

  <parameters name="γ" type="CONSTANT"
      expression="0.5797"
      description="Daily rate of progression through infectious stages (I1→I2 and I2→R)."
      unit="1/day"/>

  <parameters name="π" type="CONSTANT"
      description="Initial susceptibility: proportion of population initially susceptible (S(0)=πN)."
      unit="dimensionless"/>

  <parameters name="λ₀" type="CONSTANT"
      description="Initial infection rate / initial force of infection at t=0 (EGR parametrization)."
      unit="1/day"/>

  <parameters name="ψ" type="CONSTANT"
      description="Exponential growth rate (EGR) at epidemic start (used for re-parameterization)."
      unit="1/day"/>

  <parameters name="N" type="CONSTANT"
      description="Population size for the season (fixed within season)."
      unit="people"/>

  <parameters name="N2012/13" type="CONSTANT"
      expression="53679750"
      description="Population size for season 2012/13."
      unit="people"/>

  <parameters name="N2013/14" type="CONSTANT"
      expression="54091200"
      description="Population size for season 2013/14."
      unit="people"/>

  <parameters name="N2014/15" type="CONSTANT"
      expression="54551450"
      description="Population size for season 2014/15."
      unit="people"/>

  <parameters name="pICU" type="CONSTANT"
      description="Probability of ICU/HDU admission given infection (observation/severity link)."
      unit="dimensionless"/>

  <parameters name="μICU|E" type="CONSTANT"
      expression="5.708"
      description="Mean time from infection to ICU/HDU admission (Gamma waiting time)."
      unit="day"/>

  <parameters name="σ²ICU|E" type="CONSTANT"
      expression="18.24"
      description="Variance of time from infection to ICU/HDU admission (Gamma waiting time)."
      unit="day^2"/>

  <parameters name="η" type="CONSTANT"
      description="Negative binomial over-dispersion for weekly ICU/HDU counts."
      unit="dimensionless"/>

  <!-- Derived/summary quantities mentioned in the paper -->
  <parameters name="R₀" type="EXPRESSION"
      description="Basic reproduction number (derived; depends on ψ, σ, γ under EGR parametrization)."
      unit="dimensionless"/>

  <parameters name="Rₙ" type="EXPRESSION"
      description="Effective reproduction number in partially susceptible population (function of π and ψ)."
      unit="dimensionless"/>

  <parameters name="TAR" type="EXPRESSION"
      description="Total attack rate over the season (probability a susceptible individual is infected)."
      unit="dimensionless"/>

  <parameters name="I₀" type="EXPRESSION"
      description="Initial total infectious individuals (derived quantity in EGR parametrization)."
      unit="people"/>

  <!-- =========================================================
       Compartments (SEIR with 2-stage E and 2-stage I, as in Fig.2 / Eq.(2))
       ========================================================= -->
  <compartments PrimaryName="Susceptible" SecondaryName="All"
      population="0"
      description="S(t): susceptible individuals." >
    <!-- S -> E1 via contact with infectious (frequency-dependent); paper: λ(t) S -->
    <outgoingFlows xsi:type="seir:ContactFlow"
        contactRateParameter="//@parameters.0"
        contactCompartment="//@compartments.4"
        target="//@compartments.1"
        description="Infection: Susceptible → Exposed stage 1 (S→E1) via contact with infectious (λ(t)=β I/N)."/>
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="Stage 1"
      population="0"
      description="E1(t): first exposed (latent) stage." >
    <!-- E1 -> E2 at rate σ -->
    <outgoingFlows xsi:type="seir:RateFlow"
        rateParameter="//@parameters.3"
        target="//@compartments.2"
        description="Latent progression: E1 → E2 at rate σ."/>
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="Stage 2"
      population="0"
      description="E2(t): second exposed (latent) stage." >
    <!-- E2 -> I1 at rate σ -->
    <outgoingFlows xsi:type="seir:RateFlow"
        rateParameter="//@parameters.3"
        target="//@compartments.3"
        description="Onset of infectiousness: E2 → I1 at rate σ."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Stage 1"
      population="0"
      description="I1(t): first infectious stage." >
    <!-- I1 -> I2 at rate γ -->
    <outgoingFlows xsi:type="seir:RateFlow"
        rateParameter="//@parameters.4"
        target="//@compartments.4"
        description="Infectious progression: I1 → I2 at rate γ."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Stage 2"
      population="0"
      description="I2(t): second infectious stage. Paper defines I(t)=I1(t)+I2(t) for force of infection." >
    <!-- I2 -> R at rate γ -->
    <outgoingFlows xsi:type="seir:RateFlow"
        rateParameter="//@parameters.4"
        target="//@compartments.5"
        description="Recovery/removal: I2 → Removed at rate γ."/>
  </compartments>

  <compartments PrimaryName="Removed" SecondaryName="All"
      population="0"
      description="R(t): removed/recovered individuals (no longer infectious; assumed immune for season)." />

  <!-- =========================================================
       Notes on observation model (ICU admissions):
       The paper links infections to ICU admissions via:
         - pICU (probability ICU given infection)
         - Gamma waiting time from infection to ICU with mean μICU|E and variance σ²ICU|E
         - Weekly counts ICU_w ~ NegBin(μ_w, η)
       This is not represented as compartments/flows in the SEIR metamodel (it is an observation process),
       so parameters are included but no extra compartments are added.
       ========================================================= -->

</seir:SEIRModel>
