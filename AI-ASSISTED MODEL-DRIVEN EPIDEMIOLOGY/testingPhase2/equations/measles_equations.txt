\[
\frac{dS_C}{dt}=\Lambda-(\mu+\xi_M+\xi_U+\psi)S_C-\beta_C(I_C+I_A)S_C,
\]
\[
\frac{dS_A}{dt}=\psi\,S_C-\beta_A(I_C+I_A)S_A-\mu S_A-\xi_A S_A,
\]
\[
\frac{dU_C^{V}}{dt}=\xi_U S_C-\epsilon\beta_C(I_C+I_A)U_C^{V}-(\mu+\eta_U+\psi)U_C^{V},
\]
\[
\frac{dU_A^{V}}{dt}=\psi U_C^{V}-\eta_A U_A^{V}-\beta_A\epsilon(I_C+I_A)U_A^{V}-\mu U_A^{V},
\]
\[
\frac{dV_A}{dt}=\xi_A S_A-V_A(\mu+\eta_A),
\]
\[
\frac{dM_C^{V}}{dt}=\xi_M S_C-(\mu+\eta_M)M_C^{V},
\]
\[
\frac{dE_C}{dt}=\beta_C(I_C+I_A)S_C+\beta_C\epsilon(I_C+I_A)U_C^{V}-(\delta+\mu+\psi)E_C,
\]
\[
\frac{dE_A}{dt}=\beta_A(I_C+I_A)S_A+\beta_A\epsilon(I_C+I_A)U_A^{V}+\psi E_C-(\delta+\mu)E_A,
\]
\[
\frac{dI_C}{dt}=\delta E_C-(\gamma+d+\mu+\psi)I_C,
\]
\[
\frac{dI_A}{dt}=\delta E_A+\psi E_C-(\gamma+d+\mu)I_A,
\]
\[
\frac{dI_{mC}}{dt}=\gamma I_C+\eta_M M_C^{V}-(\mu+\psi)I_{mC}+\eta_U U_C^{V},
\]
\[
\frac{dI_{mA}}{dt}=\eta_A(U_A^{V}+V_A)+\gamma I_A-\mu I_{mA}+\psi I_{mC}.
\]
