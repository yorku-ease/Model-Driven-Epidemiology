\[
\frac{\partial S_h(a,t)}{\partial a} + \frac{\partial S_h(a,t)}{\partial t}
= -\,\beta_h(a) I_v(t)\frac{S_h(a,t)}{N_h(t)} - \mu_h S_h(a,t),
\]
\[
\frac{\partial E_h(a,t)}{\partial a} + \frac{\partial E_h(a,t)}{\partial t}
= \beta_h(a) I_v(t)\frac{S_h(a,t)}{N_h(t)} - \big(\mu_h + k(a)\big)E_h(a,t),
\]
\[
\frac{\partial I_h(a,t)}{\partial a} + \frac{\partial I_h(a,t)}{\partial t}
= \rho k(a)E_h(a,t) + \sigma\beta_h(a) I_v(t)\frac{A_h(a,t)}{N_h(t)} - \big(\mu_h + \gamma_1(a)\big)I_h(a,t),
\]
\[
\frac{\partial A_h(a,t)}{\partial a} + \frac{\partial A_h(a,t)}{\partial t}
= (1-\rho)k(a)E_h(a,t) - \sigma\beta_h(a) I_v(t)\frac{A_h(a,t)}{N_h(t)} - \big(\mu_h + \gamma_2(a)\big)A_h(a,t),
\]
\[
\frac{\partial R_h(a,t)}{\partial a} + \frac{\partial R_h(a,t)}{\partial t}
= \gamma_1(a)I_h(a,t) + \gamma_2(a)A_h(a,t) - \mu_h R_h(a,t),
\]
\[
\frac{dI_v(t)}{dt}
= (1-I_v(t))\frac{1}{N_h(t)}\int_{0}^{\infty}\beta_v(a)\big(I_h(a,t)+\sigma A_h(a,t)\big)\,da - \mu_v I_v(t),
\]
