    <outgoingFlows xsi:type="seir:ContactFlow" contactRate="0.0" contactRateParameter="//@parameters.4"

    <outgoingFlows xsi:type="seir:ContactFlow" contactRate="0.0" contactRateParameter="//@parameters.3"

    <outgoingFlows xsi:type="seir:RateFlow" rate="0.0" rateParameter="//@parameters.7"

  </compartments>
