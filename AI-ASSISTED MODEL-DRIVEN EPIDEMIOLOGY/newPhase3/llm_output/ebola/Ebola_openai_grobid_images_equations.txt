<?xml version="1.0" encoding="UTF-8"?>
<seir:SEIRModel xmi:version="2.0"
    xmlns:xmi="http://www.omg.org/XMI"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xmlns:seir="http://example.com/seirmodel">

  <!-- ===================== -->
  <!-- Parameters (symbolic) -->
  <!-- ===================== -->
  <parameters name="α₁" description="Recruitment/inflow into susceptible humans (birth/entry)" unit="1/time"/>
  <parameters name="α₂" description="Recruitment/inflow into infected humans (importation/recruitment of infected)" unit="1/time"/>

  <parameters name="ψ₁" description="Effective contact/transmission coefficient from infected humans (P) to susceptible humans (A)" unit="1/(population·time)"/>
  <parameters name="ψ₂" description="Effective contact/transmission coefficient from environmental pathogen load (U) to susceptible humans (A)" unit="1/(environment·time)"/>
  <parameters name="ψ₃" description="Effective contact/transmission coefficient from deceased humans (S) to susceptible humans (A) (e.g., unsafe burial contact)" unit="1/(population·time)"/>

  <parameters name="ε" description="Natural death/removal rate of humans (applies to A, P, F, K; also appears in death contributions to S)" unit="1/time"/>

  <parameters name="ρ₁" description="Disease-induced death rate for infected humans (P)" unit="1/time"/>
  <parameters name="ρ₂" description="Shedding/contamination rate from infected humans (P) to environment (U)" unit="1/time"/>
  <parameters name="ρ₃" description="Quarantine/isolation rate of infected humans (P) into quarantined class (F)" unit="1/time"/>

  <parameters name="ω₁" description="Recovery rate of quarantined humans (F) to recovered (K)" unit="1/time"/>
  <parameters name="ω₂" description="Disease-induced death rate for quarantined humans (F)" unit="1/time"/>

  <parameters name="h" description="Contamination/shedding rate from deceased humans (S) to environment (U)" unit="1/time"/>
  <parameters name="c" description="Burial/removal rate of deceased humans (S) (safe burial/clearance)" unit="1/time"/>

  <parameters name="b" description="Environmental decay/clearance rate of Ebola pathogens from environment (U)" unit="1/time"/>
  <parameters name="d" description="External input/source term into environment (U) (e.g., bush meat/environmental introduction)" unit="environment/time"/>

  <!-- Expression parameters to preserve exact ODE structure where needed -->
  <parameters name="λ(A,P,S,U)" type="EXPRESSION"
              expression="(ψ₁*P + ψ₃*S + ψ₂*U)"
              description="Force of infection multiplier acting on susceptible humans (A)" unit="1/time"/>

  <parameters name="δ_P" type="EXPRESSION"
              expression="(ρ₁ + ρ₂ + ρ₃ + ε)"
              description="Total removal rate from infected humans (P)" unit="1/time"/>

  <parameters name="δ_F" type="EXPRESSION"
              expression="(ω₁ + ω₂ + ε)"
              description="Total removal rate from quarantined humans (F)" unit="1/time"/>

  <parameters name="δ_S" type="EXPRESSION"
              expression="(h + c)"
              description="Total removal rate from deceased humans (S) (to environment via h and burial via c)" unit="1/time"/>

  <!-- ===================== -->
  <!-- Compartments          -->
  <!-- ===================== -->
  <compartments PrimaryName="Susceptible" SecondaryName="Humans (A)" population="0.0">
    <!-- Infection incidence: A -> P with rate A*(ψ₁P + ψ₃S + ψ₂U).
         Metamodel ContactFlow supports dependence on ONE contact compartment; we represent the three pathways explicitly. -->

    <!-- Direct transmission from infected humans -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.2"
                   contactCompartment="//@compartments.1"
                   target="//@compartments.1"
                   description="Infection via direct contact with infected humans: ψ₁*A*P"/>

    <!-- Transmission from deceased humans (unsafe burial contact) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.4"
                   contactCompartment="//@compartments.5"
                   target="//@compartments.1"
                   description="Infection via contact with deceased humans: ψ₃*A*S"/>

    <!-- Transmission from contaminated environment -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.3"
                   contactCompartment="//@compartments.4"
                   target="//@compartments.1"
                   description="Infection via contaminated environment: ψ₂*A*U"/>

    <!-- Natural death/removal of susceptible humans -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.5"
                   target="//@compartments.5"
                   description="Natural death of susceptible humans (modeled as removal into deceased class at rate ε*A)"/>
  </compartments>

  <compartments PrimaryName="Infected" SecondaryName="Humans (P)" population="0.0">
    <!-- Quarantine -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.8"
                   target="//@compartments.2"
                   description="Quarantine/isolation of infected humans at rate ρ₃*P"/>

    <!-- Shedding to environment -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.7"
                   target="//@compartments.4"
                   description="Shedding/contamination from infected humans to environment at rate ρ₂*P"/>

    <!-- Death of infected (disease-induced + natural) -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.14"
                   target="//@compartments.5"
                   description="Death/removal of infected humans at rate (ε+ρ₁)*P (contributes to deceased class)"/>
  </compartments>

  <compartments PrimaryName="Quarantined" SecondaryName="Humans (F)" population="0.0">
    <!-- Recovery -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.9"
                   target="//@compartments.3"
                   description="Recovery of quarantined humans at rate ω₁*F (permanent immunity)"/>

    <!-- Death of quarantined (disease-induced + natural) -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.15"
                   target="//@compartments.5"
                   description="Death/removal of quarantined humans at rate (ε+ω₂)*F (contributes to deceased class)"/>
  </compartments>

  <compartments PrimaryName="Recovered" SecondaryName="Humans (K)" population="0.0">
    <!-- Natural death/removal of recovered humans -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.5"
                   target="//@compartments.5"
                   description="Natural death of recovered humans at rate ε*K"/>
  </compartments>

  <compartments PrimaryName="Environment" SecondaryName="Ebola pathogen load (U)" population="0.0">
    <!-- Environmental decay/clearance -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.12"
                   target="//@compartments.4"
                   description="Environmental decay/clearance of pathogen load at rate b*U (modeled as self-loop sink)"/>
  </compartments>

  <compartments PrimaryName="Deceased" SecondaryName="Humans (S)" population="0.0">
    <!-- Contamination from corpses to environment -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.11"
                   target="//@compartments.4"
                   description="Contamination from deceased humans to environment at rate h*S"/>

    <!-- Burial/removal of corpses -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.10"
                   target="//@compartments.5"
                   description="Safe burial/removal of deceased humans at rate c*S (modeled as self-loop sink)"/>
  </compartments>

  <!-- ===================== -->
  <!-- External sources/sinks -->
  <!-- ===================== -->

  <!-- Recruitment into Susceptible (α₁) -->
  <birthSources name="Recruitment_Susceptible"
                rateParameter="//@parameters.0"
                targetCompartment="//@compartments.0"/>

  <!-- Recruitment/importation into Infected (α₂) -->
  <birthSources name="Recruitment_Infected"
                rateParameter="//@parameters.1"
                targetCompartment="//@compartments.1"/>

  <!-- External input into Environment (d) -->
  <birthSources name="External_Environment_Input"
                rateParameter="//@parameters.13"
                targetCompartment="//@compartments.4"/>

  <!-- Note: The model equations treat environmental decay (-bU) and corpse burial (-cS) as removals to outside.
       Because the metamodel requires a target compartment for flows, these are represented above as self-loop sinks
       with descriptive labels. If your toolchain supports explicit sinks, replace those self-loops with DeathSink-like
       constructs for non-human quantities. -->

</seir:SEIRModel>
