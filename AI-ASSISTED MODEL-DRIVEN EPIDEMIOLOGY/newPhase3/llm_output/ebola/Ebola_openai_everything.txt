  <!-- ===================== -->
  <!-- Parameters (symbols, descriptions, values when provided) -->

  <!-- ===================== -->
  <parameters name="α₁" expression="10" description="Recruitment rate into susceptible people A" unit="per day"/>
  <parameters name="α₂" expression="3" description="Recruitment/entry rate into infected people P" unit="per day"/>

  <parameters name="ψ₁" expression="0.006" description="Effective contact rate with infectious humans (drives A→P transmission via P)" unit="1/(person·day)"/>
  <parameters name="ψ₂" expression="0.0012" description="Contact rate with Ebola virus pathogens in the environment (drives A→P via U)" unit="1/(pathogen·day)"/>
  <parameters name="ψ₃" expression="0.006" description="Effective contact rate with deceased humans (drives A→P via S)" unit="1/(person·day)"/>

  <parameters name="ε" expression="0.5" description="Natural death/removal rate of humans (applies to A,P,F,K in equations)" unit="1/day"/>

  <parameters name="ρ₁" expression="0.123" description="Disease-induced death rate of infected humans (contributes to S from P)" unit="1/day"/>
  <parameters name="ρ₂" expression="0.04" description="Shedding rate of infected humans into environment (P→U contribution)" unit="1/day"/>
  <parameters name="ρ₃" expression="0.12" description="Rate at which infected people are quarantined (P→F)" unit="1/day"/>

  <parameters name="ω₁" expression="0.04" description="Recovery rate of quarantined people (F→K contribution)" unit="1/day"/>
  <parameters name="ω₂" expression="0.22" description="Death rate in quarantine due to infection (contributes to S from F)" unit="1/day"/>

  <parameters name="h" expression="0.04" description="Shedding rate from deceased people into environment (S→U contribution)" unit="1/day"/>
  <parameters name="c" expression="0.8" description="Burial/removal rate of deceased people (removes S)" unit="1/day"/>

  <parameters name="b" expression="0" description="Clearance/decay rate of Ebola virus pathogens in the environment (removes U)" unit="1/day"/>
  <parameters name="d" expression="0.0015" description="External input/source term into environment U" unit="pathogen/day"/>

  <!-- ===================== -->
  <!-- Compartments -->
  <!-- Order follows the paper's state vector (A,P,F,K,U,S) -->
  <!-- ===================== -->
  <compartments PrimaryName="Susceptible" SecondaryName="A" population="0.9">
    <!-- Infection incidence: A -> P driven by contact with P, S, U -->
    <outgoingFlows xsi:type="seir:ContactFlow" contactRateParameter="//@parameters.2"
                   contactCompartment="//@compartments.1" target="//@compartments.1"
                   description="Infection of susceptible by infectious humans: ψ₁ * A * P"/>
    <outgoingFlows xsi:type="seir:ContactFlow" contactRateParameter="//@parameters.4"
                   contactCompartment="//@compartments.4" target="//@compartments.1"
                   description="Infection of susceptible by environmental pathogens: ψ₂ * A * U"/>
    <outgoingFlows xsi:type="seir:ContactFlow" contactRateParameter="//@parameters.5"
                   contactCompartment="//@compartments.5" target="//@compartments.1"
                   description="Infection of susceptible by deceased humans: ψ₃ * A * S"/>

    <!-- Natural death/removal from A (equation term: -εA) -->
    <outgoingFlows xsi:type="seir:RateFlow" rateParameter="//@parameters.6" target="//@compartments.0"
                   description="Natural death/removal from Susceptible (modeled as sink; metamodel lacks explicit sink compartment)."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="P" population="0.8">
    <!-- Quarantine: P -> F (equation term: +ρ₃P in F, -ρ₃P in P) -->
    <outgoingFlows xsi:type="seir:RateFlow" rateParameter="//@parameters.9" target="//@compartments.2"
                   description="Quarantine of infectious individuals: ρ₃ * P"/>

    <!-- Disease-induced death contributes to S: (ε+ρ₁)P appears in dS/dζ -->
    <outgoingFlows xsi:type="seir:RateFlow" rateParameter="//@parameters.7" target="//@compartments.5"
                   description="Disease-induced death from infectious to deceased: ρ₁ * P"/>
    <outgoingFlows xsi:type="seir:RateFlow" rateParameter="//@parameters.6" target="//@compartments.5"
                   description="Natural death/removal of infectious contributing to deceased term in paper: ε * P"/>

    <!-- Environmental shedding is not a person-transfer; represented as a flow to Environment compartment -->
    <outgoingFlows xsi:type="seir:RateFlow" rateParameter="//@parameters.8" target="//@compartments.4"
                   description="Shedding from infectious humans into environment: ρ₂ * P"/>

    <!-- Additional removal terms in dP/dζ: -(ρ₂+ε)P are already represented above (ρ₂ to U, ε to S),
         and -(ρ₃)P to F. -->
  </compartments>

  <compartments PrimaryName="Quarantined" SecondaryName="F" population="0.1">
    <!-- Recovery contribution: +ω₁F in dK/dζ -->
    <outgoingFlows xsi:type="seir:RateFlow" rateParameter="//@parameters.10" target="//@compartments.3"
                   description="Recovery of quarantined individuals: ω₁ * F"/>

    <!-- Death contribution: (ω₂+ε)F in dS/dζ -->
    <outgoingFlows xsi:type="seir:RateFlow" rateParameter="//@parameters.11" target="//@compartments.5"
                   description="Disease-induced death in quarantine to deceased: ω₂ * F"/>
    <outgoingFlows xsi:type="seir:RateFlow" rateParameter="//@parameters.6" target="//@compartments.5"
                   description="Natural death/removal of quarantined contributing to deceased term in paper: ε * F"/>
  </compartments>

  <compartments PrimaryName="Recovered" SecondaryName="K" population="0.2">
    <!-- Natural death/removal from K (equation term: -εK) -->
    <outgoingFlows xsi:type="seir:RateFlow" rateParameter="//@parameters.6" target="//@compartments.3"
                   description="Natural death/removal from Recovered (modeled as sink; metamodel lacks explicit sink compartment)."/>
  </compartments>

  <compartments PrimaryName="Environment" SecondaryName="U" population="0">
    <!-- Environmental decay/clearance: -bU -->
    <outgoingFlows xsi:type="seir:RateFlow" rateParameter="//@parameters.14" target="//@compartments.4"
                   description="Environmental clearance/decay of pathogens (modeled as sink; metamodel lacks explicit sink compartment)."/>
  </compartments>

  <compartments PrimaryName="Deceased" SecondaryName="S" population="0.5">
    <!-- Shedding from deceased into environment: +hS in dU/dζ -->
    <outgoingFlows xsi:type="seir:RateFlow" rateParameter="//@parameters.12" target="//@compartments.4"
                   description="Shedding from deceased into environment: h * S"/>

    <!-- Burial/removal: -(h+c)S includes removal cS; hS is represented above as S->U -->
    <outgoingFlows xsi:type="seir:RateFlow" rateParameter="//@parameters.13" target="//@compartments.5"
                   description="Burial/removal of deceased (modeled as sink; metamodel lacks explicit sink compartment): c * S"/>
  </compartments>

  <!-- ===================== -->
  <!-- Birth / recruitment sources (demographic inflows) -->
  <!-- Equations: +α₁ into A, +α₂ into P -->
  <!-- ===================== -->
  <birthSources name="Recruitment_to_Susceptible" rateParameter="//@parameters.0" targetCompartment="//@compartments.0"/>
  <birthSources name="Recruitment_to_Infectious" rateParameter="//@parameters.1" targetCompartment="//@compartments.1"/>

  <!-- NOTE on environment input d:
       The metamodel has no generic "source term" for non-population compartments except BirthSource.
       We encode d as a BirthSource into Environment. -->
  <birthSources name="External_Input_to_Environment" rateParameter="//@parameters.15" targetCompartment="//@compartments.4"/>

</seir:SEIRModel>
