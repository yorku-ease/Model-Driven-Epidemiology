<?xml version="1.0" encoding="UTF-8"?>
<seir:SEIRModel xmi:version="2.0"
    xmlns:xmi="http://www.omg.org/XMI"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xmlns:seir="http://example.com/seirmodel">

  <!-- ===================== -->
  <!-- Parameters (paper Table 1 + equations) -->
  <!-- ===================== -->
  <parameters name="A" type="CONSTANT"
              expression="1000"
              description="Recruitment rate of humans (all newborns enter Susceptible)"
              unit="humans/time"/>
  <parameters name="β" type="CONSTANT"
              expression="0.01"
              description="Transmission coefficient / probability of successful infection (primary infection term β S I). Table gives range [0.01, 5]."
              unit="1/(human·time)"/>
  <parameters name="ε" type="CONSTANT"
              expression="0.0"
              description="Correction parameter for reinfection term (reinfection force ε β R I), 0–1"
              unit="dimensionless"/>
  <parameters name="u" type="CONSTANT"
              expression="0.0"
              description="Vaccination rate moving Susceptible to Vaccinated, 0–1"
              unit="1/time"/>
  <parameters name="θ" type="CONSTANT"
              expression="0.067"
              description="Vaccine waning rate moving Vaccinated to Susceptible. Table gives range [0.067, 0.1]."
              unit="1/time"/>
  <parameters name="γ" type="CONSTANT"
              expression="0.5"
              description="Recovery rate moving Infected to Recovered. Table gives range [0.5, 2]."
              unit="1/time"/>
  <parameters name="δ" type="CONSTANT"
              expression="0.003"
              description="Waning natural immunity rate moving Recovered to Susceptible. Table gives range [0.003, 0.02]."
              unit="1/time"/>
  <parameters name="μ" type="CONSTANT"
              expression="0.0"
              description="Natural death rate (applies to all compartments). Table indicates estimated ~ 1/(65×12 months); numeric not cleanly extractable from table."
              unit="1/time"/>

  <!-- Dimensionless / derived symbols mentioned in text (kept as EXPRESSION parameters) -->
  <parameters name="N" type="EXPRESSION"
              expression="S+V+I+R"
              description="Total population size"
              unit="humans"/>
  <parameters name="b" type="EXPRESSION"
              expression="β*N/γ"
              description="Dimensionless transmission scaling used in nondimensional system (paper Section 3.2)"
              unit="dimensionless"/>
  <parameters name="m" type="EXPRESSION"
              expression="μ/γ"
              description="Dimensionless natural death rate used in nondimensional system"
              unit="dimensionless"/>
  <parameters name="c" type="EXPRESSION"
              expression="δ/γ"
              description="Dimensionless immunity waning rate used in nondimensional system"
              unit="dimensionless"/>
  <parameters name="d" type="EXPRESSION"
              expression="θ/γ"
              description="Dimensionless vaccine waning rate used in nondimensional system"
              unit="dimensionless"/>
  <parameters name="e" type="EXPRESSION"
              expression="ε"
              description="Dimensionless reinfection correction parameter used in nondimensional system"
              unit="dimensionless"/>
  <parameters name="ū" type="EXPRESSION"
              expression="u/γ"
              description="Dimensionless vaccination rate used in nondimensional system"
              unit="dimensionless"/>

  <!-- Reproduction number and bifurcation helper quantity (symbols mentioned) -->
  <parameters name="ℛ_c" type="EXPRESSION"
              expression="(b*(d+m))/((d+m+ū)*(1+m))"
              description="Control reproduction number as given in the paper (dimensionless form)"
              unit="dimensionless"/>
  <parameters name="K" type="EXPRESSION"
              expression="(m+c+1)*(m+d) - e*(m+d+ū)"
              description="Bifurcation sign quantity (paper uses calligraphic K / script K) determining backward vs forward bifurcation"
              unit="dimensionless"/>

  <!-- ===================== -->
  <!-- Compartments (system (1)) -->
  <!-- ===================== -->
  <compartments PrimaryName="Susceptible" population="0">
    <!-- Infection: S -> I via contact with Infected at rate β S I -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.1"
                   contactCompartment="//@compartments.2"
                   target="//@compartments.2"
                   description="Primary infection of susceptible individuals via contact with infected (β S I)"/>
    <!-- Vaccination: S -> V at rate u S -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.3"
                   target="//@compartments.1"
                   description="Vaccination of susceptible individuals (u S)"/>
  </compartments>

  <compartments PrimaryName="Vaccinated" population="0">
    <!-- Vaccine waning: V -> S at rate θ V -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.4"
                   target="//@compartments.0"
                   description="Waning of vaccine-induced protection (θ V)"/>
  </compartments>

  <compartments PrimaryName="Infected" SecondaryName="Tuberculosis infectious" population="0">
    <!-- Recovery: I -> R at rate γ I -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.5"
                   target="//@compartments.3"
                   description="Recovery from active TB infection (γ I)"/>
  </compartments>

  <compartments PrimaryName="Recovered" population="0">
    <!-- Reinfection: R -> I via contact with Infected at rate ε β R I -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.1"
                   contactCompartment="//@compartments.2"
                   target="//@compartments.2"
                   description="Reinfection of recovered individuals via contact with infected (ε β R I); implemented as β contact flow with ε handled as multiplier in simulation/expression"/>
    <!-- Loss of natural immunity: R -> S at rate δ R -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.6"
                   target="//@compartments.0"
                   description="Waning of natural immunity (δ R)"/>
  </compartments>

  <!-- ===================== -->
  <!-- Demography: births and natural deaths (system (1)) -->
  <!-- ===================== -->
  <birthSources name="Recruitment"
                rateParameter="//@parameters.0"
                targetCompartment="//@compartments.0"/>

  <deathSinks name="NaturalDeath_S"
              rateParameter="//@parameters.7"
              sourceCompartment="//@compartments.0"/>
  <deathSinks name="NaturalDeath_V"
              rateParameter="//@parameters.7"
              sourceCompartment="//@compartments.1"/>
  <deathSinks name="NaturalDeath_I"
              rateParameter="//@parameters.7"
              sourceCompartment="//@compartments.2"/>
  <deathSinks name="NaturalDeath_R"
              rateParameter="//@parameters.7"
              sourceCompartment="//@compartments.3"/>

</seir:SEIRModel>
