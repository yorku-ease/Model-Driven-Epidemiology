<?xml version="1.0" encoding="UTF-8"?>
<seir:SEIRModel xmi:version="2.0"
    xmlns:xmi="http://www.omg.org/XMI"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xmlns:seir="http://example.com/seirmodel">

  <!-- ===================== -->
  <!-- Parameters (paper Eq. (1), Table 1) -->
  <!-- ===================== -->
  <parameters name="A" expression="1000"
              description="Recruitment rate of human (all newborn are susceptible)"
              unit="human/time"/>
  <parameters name="β" type="CONSTANT" expression="0.01..5"
              description="Probability/rate of successful infection (primary infection term β S I)"
              unit="1/(human·time)"/>
  <parameters name="ε" type="CONSTANT" expression="0..1"
              description="Correction parameter for reinfection (reinfection force ε β R I)"
              unit="dimensionless"/>
  <parameters name="u" type="CONSTANT" expression="0..1"
              description="Rate of vaccination (Susceptible → Vaccinated)"
              unit="1/time"/>
  <parameters name="θ" type="CONSTANT" expression="0.067..0.1"
              description="Vaccine wane rate (Vaccinated → Susceptible)"
              unit="1/time"/>
  <parameters name="γ" type="CONSTANT" expression="0.5..2"
              description="Recovery rate (Infected → Recovered)"
              unit="1/time"/>
  <parameters name="δ" type="CONSTANT" expression="0.003..0.02"
              description="Drop out rate due to waning immunity (Recovered → Susceptible)"
              unit="1/time"/>
  <parameters name="μ" type="CONSTANT" expression="estimated"
              description="Natural death rate (applies to all compartments)"
              unit="1/time"/>

  <!-- ===================== -->
  <!-- Compartments (paper: S, V, I, R) -->
  <!-- ===================== -->
  <compartments PrimaryName="Susceptible" population="0">
    <!-- Infection: S -> Infected via contact with Infected at rate β*S*I -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.1"
                   contactCompartment="//@compartments.2"
                   target="//@compartments.2"
                   description="Primary infection: Susceptible becomes Infected through contact with Infected (β S I)"/>
    <!-- Vaccination: S -> Vaccinated at rate u*S -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.3"
                   target="//@compartments.1"
                   description="Vaccination of susceptible individuals (u S)"/>
  </compartments>

  <compartments PrimaryName="Vaccinated" population="0">
    <!-- Vaccine waning: V -> Susceptible at rate θ*V -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.4"
                   target="//@compartments.0"
                   description="Vaccine waning: vaccinated lose protection and return to Susceptible (θ V)"/>
  </compartments>

  <compartments PrimaryName="Infected" population="0">
    <!-- Recovery: I -> Recovered at rate γ*I -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.5"
                   target="//@compartments.3"
                   description="Recovery from TB infection (γ I)"/>
  </compartments>

  <compartments PrimaryName="Recovered" population="0">
    <!-- Reinfection: R -> Infected via contact with Infected at rate εβ*R*I -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.1"
                   contactCompartment="//@compartments.2"
                   target="//@compartments.2"
                   description="Reinfection: recovered individuals become Infected through contact with Infected (ε β R I)">
      <stratumSpecificRates stratum="default" multiplierParameter="//@parameters.2"/>
    </outgoingFlows>

    <!-- Waning natural immunity: R -> Susceptible at rate δ*R -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.6"
                   target="//@compartments.0"
                   description="Waning immunity: recovered lose immunity and return to Susceptible (δ R)"/>
  </compartments>

  <!-- ===================== -->
  <!-- Birth / Recruitment source (paper Eq. (1a): +A into S) -->
  <!-- ===================== -->
  <birthSources name="HumanRecruitment"
                rateParameter="//@parameters.0"
                targetCompartment="//@compartments.0"
                fixedRate="true"/>

  <!-- ===================== -->
  <!-- Natural death sinks (paper: -μ in every equation) -->
  <!-- ===================== -->
  <deathSinks name="NaturalDeath_Susceptible"
              rateParameter="//@parameters.7"
              sourceCompartment="//@compartments.0"/>
  <deathSinks name="NaturalDeath_Vaccinated"
              rateParameter="//@parameters.7"
              sourceCompartment="//@compartments.1"/>
  <deathSinks name="NaturalDeath_Infected"
              rateParameter="//@parameters.7"
              sourceCompartment="//@compartments.2"/>
  <deathSinks name="NaturalDeath_Recovered"
              rateParameter="//@parameters.7"
              sourceCompartment="//@compartments.3"/>

</seir:SEIRModel>
