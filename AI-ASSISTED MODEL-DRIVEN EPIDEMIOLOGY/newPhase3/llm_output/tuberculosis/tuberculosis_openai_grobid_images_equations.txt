<?xml version="1.0" encoding="UTF-8"?>
<seir:SEIRModel xmi:version="2.0"
    xmlns:xmi="http://www.omg.org/XMI"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xmlns:seir="http://example.com/seirmodel">

  <!-- ===================== -->
  <!-- Parameters (symbolic) -->
  <!-- ===================== -->
  <parameters name="A" type="CONSTANT"
              description="Recruitment (birth/immigration) rate into Susceptible"
              unit="individuals/time"/>
  <parameters name="β" type="CONSTANT"
              description="Transmission coefficient (mass-action) for infection from Infectious"
              unit="1/(individual*time)"/>
  <parameters name="u" type="CONSTANT"
              description="Vaccination rate of Susceptible individuals (S→V)"
              unit="1/time"/>
  <parameters name="θ" type="CONSTANT"
              description="Waning rate of vaccine-induced protection (V→S)"
              unit="1/time"/>
  <parameters name="γ" type="CONSTANT"
              description="Recovery rate (I→R)"
              unit="1/time"/>
  <parameters name="δ" type="CONSTANT"
              description="Waning rate of natural (post-infection) immunity (R→S)"
              unit="1/time"/>
  <parameters name="μ" type="CONSTANT"
              description="Natural death rate (applies to all compartments)"
              unit="1/time"/>
  <parameters name="ε" type="CONSTANT"
              description="Relative susceptibility (reinfection modifier) for Recovered individuals (R→I) via εβRI"
              unit="dimensionless"/>

  <!-- ===================== -->
  <!-- Compartments (SVIR)   -->
  <!-- ===================== -->
  <compartments PrimaryName="Susceptible" SecondaryName="S"
                population="0.0">
    <!-- Infection: S -> I at rate β*S*I -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRate="0.0"
                   contactRateParameter="//@parameters.1"
                   contactCompartment="//@compartments.2"
                   target="//@compartments.2"
                   description="Primary infection of susceptible individuals (mass-action βSI)"/>
    <!-- Vaccination: S -> V at rate u*S -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rate="0.0"
                   rateParameter="//@parameters.2"
                   target="//@compartments.1"
                   description="Vaccination of susceptibles (uS)"/>
  </compartments>

  <compartments PrimaryName="Vaccinated" SecondaryName="V"
                population="0.0">
    <!-- Waning vaccine immunity: V -> S at rate θ*V -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rate="0.0"
                   rateParameter="//@parameters.3"
                   target="//@compartments.0"
                   description="Waning of vaccine-induced immunity (θV)"/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="I"
                population="0.0">
    <!-- Recovery: I -> R at rate γ*I -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rate="0.0"
                   rateParameter="//@parameters.4"
                   target="//@compartments.3"
                   description="Recovery from active TB infection (γI)"/>
  </compartments>

  <compartments PrimaryName="Recovered" SecondaryName="R"
                population="0.0">
    <!-- Reinfection: R -> I at rate (εβ)*R*I (contact with Infectious) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRate="0.0"
                   contactRateParameter="//@parameters.1"
                   contactCompartment="//@compartments.2"
                   target="//@compartments.2"
                   description="Reinfection of recovered individuals; effective contact rate scaled by ε (εβRI)"/>
    <!-- Loss of natural immunity: R -> S at rate δ*R -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rate="0.0"
                   rateParameter="//@parameters.5"
                   target="//@compartments.0"
                   description="Waning of natural immunity (δR)"/>
  </compartments>

  <!-- ===================== -->
  <!-- Births / recruitment   -->
  <!-- ===================== -->
  <birthSources name="Recruitment"
                rate="0.0"
                rateParameter="//@parameters.0"
                targetCompartment="//@compartments.0"
                fixedRate="true"/>

  <!-- ===================== -->
  <!-- Death sinks (μ in all) -->
  <!-- ===================== -->
  <deathSinks name="Natural_Death_S"
              rate="0.0"
              rateParameter="//@parameters.6"
              sourceCompartment="//@compartments.0"/>
  <deathSinks name="Natural_Death_V"
              rate="0.0"
              rateParameter="//@parameters.6"
              sourceCompartment="//@compartments.1"/>
  <deathSinks name="Natural_Death_I"
              rate="0.0"
              rateParameter="//@parameters.6"
              sourceCompartment="//@compartments.2"/>
  <deathSinks name="Natural_Death_R"
              rate="0.0"
              rateParameter="//@parameters.6"
              sourceCompartment="//@compartments.3"/>

  <!-- ===================== -->
  <!-- Notes on ε scaling     -->
  <!-- ===================== -->
  <!-- The metamodel ContactFlow supports a single contactRateParameter.
       The paper's reinfection term is εβRI. This is represented structurally
       as a ContactFlow R->I with base contactRateParameter β and the intended
       ε multiplier documented in the flow description (ε is listed as a parameter). -->

</seir:SEIRModel>
