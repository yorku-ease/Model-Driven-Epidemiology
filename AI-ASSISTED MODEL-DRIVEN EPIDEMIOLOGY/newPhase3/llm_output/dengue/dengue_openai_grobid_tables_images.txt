```xml
<?xml version="1.0" encoding="UTF-8"?>
<seir:SEIRModel
    xmi:version="2.0"
    xmlns:xmi="http://www.omg.org/XMI"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xmlns:seir="http://example.com/seirmodel">

  <!--
    Model: Li et al. (JMAA 524 (2023) 127127) age-structured dengue model with asymptomatic infection.
    Human host is age-structured in the paper (densities Sh(a,t), Eh(a,t), Ih(a,t), Ah(a,t), Rh(a,t)).
    Vector is SI with total mosquitoes rescaled to 1: Sv(t)=1-Iv(t), Iv(t).
    This XML encodes the compartmental structure and symbolic parameters; age-structure is represented
    via parameter functions (βₕ(a), k(a), γ₁(a), γ₂(a), βᵥ(a)) rather than explicit strata, because the
    metamodel stratification requires discrete groups (not continuous age).
  -->

  <parameters name="B" type="CONSTANT"
              description="Constant influx/recruitment rate of susceptible human hosts (boundary condition Sh(0,t)=B in original unnormalized system; in normalized system sh(0,t)=1)."
              unit="hosts/time"/>
  <parameters name="μₕ" type="CONSTANT"
              description="Natural death rate of human hosts."
              unit="1/time"/>
  <parameters name="μᵥ" type="CONSTANT"
              description="Natural death rate of mosquitoes."
              unit="1/time"/>

  <!-- Age-dependent (continuous-age) rate functions, kept symbolic -->
  <parameters name="βₕ(a)" type="VARIABLE"
              description="Age-specific transmission/contact rate from infected mosquitoes to susceptible humans."
              unit="1/time"/>
  <parameters name="βᵥ(a)" type="VARIABLE"
              description="Age-specific transmission rate from infectious humans (symptomatic and asymptomatic) to susceptible mosquitoes."
              unit="1/time"/>
  <parameters name="k(a)" type="VARIABLE"
              description="Age-specific progression rate from exposed to infectious (latent period is 1/k(a))."
              unit="1/time"/>
  <parameters name="γ₁(a)" type="VARIABLE"
              description="Age-specific recovery rate of symptomatic infectious humans."
              unit="1/time"/>
  <parameters name="γ₂(a)" type="VARIABLE"
              description="Age-specific recovery rate of asymptomatic infectious humans."
              unit="1/time"/>

  <parameters name="ρ" type="CONSTANT"
              description="Probability an exposed human becomes symptomatic infectious (fraction ρ; remaining 1-ρ become asymptomatic)."
              unit="dimensionless"/>
  <parameters name="σ" type="CONSTANT"
              description="Factor characterizing intensity of reinfection/secondary infection of asymptomatic infectious hosts; also scales asymptomatic contribution to mosquito infection in the paper."
              unit="dimensionless"/>

  <!-- Normalization constants used in system (2.2) -->
  <parameters name="Nₕ(t)" type="VARIABLE"
              description="Total human population size at time t (used in standard incidence terms in system (2.1))."
              unit="hosts"/>
  <parameters name="Nₕ*" type="CONSTANT"
              description="Limiting/steady total human population size used in normalized system (2.2) (denoted N*h in the paper)."
              unit="hosts"/>

  <!-- ===================== Compartments ===================== -->
  <compartments PrimaryName="Susceptible" SecondaryName="Human" population="0.0">
    <!-- Infection by infectious mosquitoes: Sh -> Eh with standard incidence βₕ(a) Iv / Nₕ (or /Nₕ* in normalized system) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.3"
                   contactCompartment="//@compartments.6"
                   target="//@compartments.1"
                   description="Primary infection of susceptible humans by infectious mosquitoes (force of infection βₕ(a)·Iᵥ / Nₕ*)."/>
    <!-- Natural death sink handled below via deathSinks -->
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="Human" population="0.0">
    <!-- Progression to symptomatic infectious: Eh -> Ih at rate ρ k(a) -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.5"
                   target="//@compartments.2"
                   description="Progression from exposed to symptomatic infectious humans; paper rate is ρ·k(a) (ρ applied externally)."/>
    <!-- Progression to asymptomatic infectious: Eh -> Ah at rate (1-ρ) k(a) -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.5"
                   target="//@compartments.3"
                   description="Progression from exposed to asymptomatic infectious humans; paper rate is (1-ρ)·k(a) (1-ρ applied externally)."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Human_Symptomatic" population="0.0">
    <!-- Recovery: Ih -> Rh at rate γ₁(a) -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.6"
                   target="//@compartments.4"
                   description="Recovery of symptomatic infectious humans at age-specific rate γ₁(a)."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Human_Asymptomatic" population="0.0">
    <!-- Reinfection/secondary infection: Ah -> Ih via mosquito bites at contact rate σ βₕ(a) Iv / Nₕ* -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.3"
                   contactCompartment="//@compartments.6"
                   target="//@compartments.2"
                   description="Secondary infection/reinfection of asymptomatic humans leading to symptomatic infection; paper term σ·βₕ(a)·Iᵥ / Nₕ*."/>
    <!-- Recovery: Ah -> Rh at rate γ₂(a) -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.7"
                   target="//@compartments.4"
                   description="Recovery of asymptomatic infectious humans at age-specific rate γ₂(a)."/>
  </compartments>

  <compartments PrimaryName="Recovered" SecondaryName="Human" population="0.0">
    <!-- No outgoing flows in the paper (no waning immunity modeled). -->
  </compartments>

  <!-- Vector compartments (mosquitoes), total scaled to 1 in the paper -->
  <compartments PrimaryName="Susceptible" SecondaryName="Mosquito" population="0.0">
    <!-- Infection of mosquitoes by biting infectious humans (symptomatic + σ*asymptomatic in paper integral) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.4"
                   contactCompartment="//@compartments.2"
                   target="//@compartments.6"
                   description="Infection of susceptible mosquitoes from infectious humans; paper force: (1-Iᵥ) * (1/Nₕ)∫βᵥ(a)[Iₕ(a,t)+σAₕ(a,t)]da. Encoded structurally as contact with infectious humans using βᵥ(a)."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Mosquito" population="0.0">
    <!-- No recovery in the paper; removal by natural death only (handled via deathSinks). -->
  </compartments>

  <!-- ===================== Birth sources ===================== -->
  <birthSources name="HumanRecruitment"
                rateParameter="//@parameters.0"
                targetCompartment="//@compartments.0"
                fixedRate="true"/>

  <!-- Mosquito recruitment is implicit in the paper due to rescaling to 1; not explicitly modeled as a birth term. -->

  <!-- ===================== Death sinks ===================== -->
  <deathSinks name="NaturalDeath_Human_S"
              rateParameter="//@parameters.1"
              sourceCompartment="//@compartments.0"/>
  <deathSinks name="NaturalDeath_Human_E"
              rateParameter="//@parameters.1"
              sourceCompartment="//@compartments.1"/>
  <deathSinks name="NaturalDeath_Human_I_sym"
              rateParameter="//@parameters.1"
              sourceCompartment="//@compartments.2"/>
  <deathSinks name="NaturalDeath_Human_I_asym"
              rateParameter="//@parameters.1"
              sourceCompartment="//@compartments.3"/>
  <deathSinks name="NaturalDeath_Human_R"
              rateParameter="//@parameters.1"
              sourceCompartment="//@compartments.4"/>

  <deathSinks name="NaturalDeath_Mosquito_S"
              rateParameter="//@parameters.2"
              sourceCompartment="//@compartments.5"/>
  <deathSinks name="NaturalDeath_Mosquito_I"
              rateParameter="//@parameters.2"
              sourceCompartment="//@compartments.6"/>

</seir:SEIRModel>
```
