```xml
<?xml version="1.0" encoding="UTF-8"?>
<seir:SEIRModel
    xmi:version="2.0"
    xmlns:xmi="http://www.omg.org/XMI"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xmlns:seir="http://example.com/seirmodel">

  <!-- =========================================================
       PARAMETERS (symbolic; values/units not provided except meanings)
       Source: Table 1 + equations + section text
       NOTE: Age-structured functions are represented as parameters with
             functional names (βₕ(a), k(a), γ₁(a), γ₂(a), βᵥ(a)).
       ========================================================= -->
  <parameters name="a" type="VARIABLE" description="Host chronological age (structuring variable)" unit="time (e.g., years)"/>
  <parameters name="t" type="VARIABLE" description="Time" unit="time"/>

  <parameters name="B" description="Constant influx (recruitment/birth) rate of human hosts into Sₕ at age a=0" unit="hosts/time"/>
  <parameters name="μₕ" description="Natural death rate of human hosts" unit="1/time"/>
  <parameters name="μᵥ" description="Natural death rate of mosquitoes" unit="1/time"/>

  <parameters name="βₕ(a)" type="EXPRESSION" expression="βₕ(a)" description="Age-specific transmission/contact rate from infectious mosquitoes to susceptible humans" unit="1/time"/>
  <parameters name="βᵥ(a)" type="EXPRESSION" expression="βᵥ(a)" description="Age-specific transmission rate from infectious humans (Iₕ and Aₕ) to susceptible mosquitoes" unit="1/time"/>

  <parameters name="k(a)" type="EXPRESSION" expression="k(a)" description="Age-specific progression rate from exposed to infectious (inverse of latent period)" unit="1/time"/>
  <parameters name="γ₁(a)" type="EXPRESSION" expression="γ₁(a)" description="Age-specific recovery rate of symptomatic infectious humans" unit="1/time"/>
  <parameters name="γ₂(a)" type="EXPRESSION" expression="γ₂(a)" description="Age-specific recovery rate of asymptomatic infectious humans" unit="1/time"/>

  <parameters name="ρ" description="Probability an exposed human becomes symptomatic infectious (Eₕ→Iₕ); (1-ρ) becomes asymptomatic (Eₕ→Aₕ)" unit="dimensionless"/>
  <parameters name="σ" description="Factor characterizing intensity of reinfection of asymptomatic infectious hosts (secondary infection modifier); also scales Aₕ contribution to mosquito infection in vector equation" unit="dimensionless"/>

  <parameters name="Nₕ(t)" type="EXPRESSION" expression="Nₕ(t)" description="Total human population size at time t (integral over ages of Sₕ+Eₕ+Iₕ+Aₕ+Rₕ)" unit="hosts"/>
  <parameters name="Iᵥ(t)" type="EXPRESSION" expression="Iᵥ(t)" description="Proportion of infectious mosquitoes at time t (mosquito population rescaled to 1)" unit="dimensionless"/>

  <!-- =========================================================
       COMPARTMENTS
       Host compartments are age-structured densities (a,t) in the paper.
       Vector compartments are proportions (t) with Sᵥ = 1 - Iᵥ.
       We include Sᵥ explicitly for topology (even though eliminated in ODE).
       ========================================================= -->
  <compartments PrimaryName="Susceptible" SecondaryName="Human (age-structured)" population="0.0">
    <!-- Infection: Sₕ -> Eₕ via infectious mosquitoes (frequency-dependent with Nₕ(t)) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.6"
                   contactCompartment="//@compartments.6"
                   target="//@compartments.1"
                   description="Primary infection of susceptible humans by infectious mosquitoes: βₕ(a) * Iᵥ(t) * Sₕ(a,t) / Nₕ(t)."/>
    <!-- Natural death sink (modeled as DeathSink below) -->
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="Human (age-structured)" population="0.0">
    <!-- Progression: Eₕ -> Iₕ -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.8"
                   target="//@compartments.2"
                   description="Progression to symptomatic infectious: ρ * k(a) * Eₕ(a,t). (Implemented as base k(a); split by ρ handled conceptually)."/>
    <!-- Progression: Eₕ -> Aₕ -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.8"
                   target="//@compartments.3"
                   description="Progression to asymptomatic infectious: (1-ρ) * k(a) * Eₕ(a,t). (Implemented as base k(a); split by (1-ρ) handled conceptually)."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Human Symptomatic (age-structured)" population="0.0">
    <!-- Recovery: Iₕ -> Rₕ -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.9"
                   target="//@compartments.4"
                   description="Recovery of symptomatic humans: γ₁(a) * Iₕ(a,t)."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Human Asymptomatic (age-structured)" population="0.0">
    <!-- Secondary infection / reinfection: Aₕ -> Iₕ driven by infectious mosquitoes -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.6"
                   contactCompartment="//@compartments.6"
                   target="//@compartments.2"
                   description="Secondary infection (reinfection) of asymptomatic humans leading to symptomatic infection: σ * βₕ(a) * Iᵥ(t) * Aₕ(a,t) / Nₕ(t). (σ is a modifier)."/>
    <!-- Recovery: Aₕ -> Rₕ -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.10"
                   target="//@compartments.4"
                   description="Recovery of asymptomatic humans: γ₂(a) * Aₕ(a,t)."/>
  </compartments>

  <compartments PrimaryName="Recovered" SecondaryName="Human (age-structured)" population="0.0">
    <!-- No outgoing flows besides natural death (DeathSink below) -->
  </compartments>

  <compartments PrimaryName="Susceptible" SecondaryName="Mosquito (proportion)" population="0.0">
    <!-- Infection of mosquitoes: Sᵥ -> Iᵥ via infectious humans (integral over age; frequency-dependent with Nₕ(t)) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.7"
                   contactCompartment="//@compartments.2"
                   target="//@compartments.6"
                   description="Mosquito infection by biting infectious humans: (1-Iᵥ(t))*(1/Nₕ(t))∫βᵥ(a)[Iₕ(a,t)+σAₕ(a,t)]da. (Contact compartment set to Iₕ; Aₕ contribution scaled by σ)."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Mosquito (proportion)" population="0.0">
    <!-- No recovery; natural death (DeathSink below). Inflow from Sᵥ via infection above. -->
  </compartments>

  <!-- =========================================================
       BIRTH / RECRUITMENT SOURCE
       Paper: boundary condition Sₕ(0,t)=B (original densities) and normalized sh(0,t)=1.
       Here we encode the demographic inflow as a BirthSource into Susceptible Human.
       ========================================================= -->
  <birthSources name="Human_Recruitment"
                rateParameter="//@parameters.2"
                targetCompartment="//@compartments.0"
                fixedRate="true"/>

  <!-- =========================================================
       DEATH SINKS (natural mortality)
       Humans: μₕ from all human compartments.
       Mosquitoes: μᵥ from Sᵥ and Iᵥ.
       ========================================================= -->
  <deathSinks name="Natural_Death_S_h" rateParameter="//@parameters.3" sourceCompartment="//@compartments.0"/>
  <deathSinks name="Natural_Death_E_h" rateParameter="//@parameters.3" sourceCompartment="//@compartments.1"/>
  <deathSinks name="Natural_Death_I_h" rateParameter="//@parameters.3" sourceCompartment="//@compartments.2"/>
  <deathSinks name="Natural_Death_A_h" rateParameter="//@parameters.3" sourceCompartment="//@compartments.3"/>
  <deathSinks name="Natural_Death_R_h" rateParameter="//@parameters.3" sourceCompartment="//@compartments.4"/>

  <deathSinks name="Natural_Death_S_v" rateParameter="//@parameters.4" sourceCompartment="//@compartments.5"/>
  <deathSinks name="Natural_Death_I_v" rateParameter="//@parameters.4" sourceCompartment="//@compartments.6"/>

  <!-- =========================================================
       NOTES / LIMITATIONS (kept as descriptions in parameters/flows):
       - The metamodel is not PDE/age-transport aware; age-structure is
         represented via SecondaryName and functional parameters.
       - The E->I and E->A split uses ρ and (1-ρ); metamodel RateFlow
         supports only one rate parameter per flow, so the split is
         documented in descriptions rather than encoded as separate
         multiplier parameters.
       - Vector equation includes factor (1-Iᵥ(t)); represented in flow
         description but not enforceable structurally in this metamodel.
       ========================================================= -->

</seir:SEIRModel>
```
