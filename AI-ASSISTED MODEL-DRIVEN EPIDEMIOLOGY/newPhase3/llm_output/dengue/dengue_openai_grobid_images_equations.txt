  <parameters name="B" type="CONSTANT" description="Influx (recruitment/birth) rate into susceptible human hosts" unit="hosts/time"/>
  <parameters name="μₕ" type="CONSTANT" description="Natural death rate of human hosts" unit="1/time"/>
  <parameters name="μᵥ" type="CONSTANT" description="Natural death rate of mosquitoes (vectors)" unit="1/time"/>

  <parameters name="βᵥ(a)" type="EXPRESSION" description="Age-dependent transmission rate from infectious humans to susceptible mosquitoes (via biting)" unit="1/time"/>

  <parameters name="k(a)" type="EXPRESSION" description="Age-dependent progression rate from exposed (latent) humans to infectious humans" unit="1/time"/>
  <parameters name="ρ" type="CONSTANT" description="Probability an exposed human becomes symptomatic infectious (0<ρ<1)" unit="dimensionless"/>
  <parameters name="σ" type="CONSTANT" description="Reinfection intensity factor for asymptomatic humans; scales secondary infection/contact term" unit="dimensionless"/>

  <parameters name="γ₁(a)" type="EXPRESSION" description="Age-specific recovery rate of symptomatic infectious humans" unit="1/time"/>
  <parameters name="γ₂(a)" type="EXPRESSION" description="Age-specific recovery rate of asymptomatic infectious humans" unit="1/time"/>

  <parameters name="Nₕ(t)" type="EXPRESSION" description="Total human population size at time t (normalizing term in standard incidence)" unit="hosts"/>
  <parameters name="∫βᵥ(a)(Iₕ(a,t)+σAₕ(a,t))da" type="EXPRESSION"
              description="Age-integrated infectiousness of humans contributing to mosquito infection" unit="1/time"/>

  <!-- Explicit numeric examples reported in the paper (used in simulations; not assumed globally) -->
  <parameters name="B_example_DFE" type="CONSTANT" expression="250" description="Example value used in a DFE global stability simulation" unit="hosts/time"/>
  <parameters name="μₕ_example_DFE" type="CONSTANT" expression="0.005" description="Example value used in a DFE global stability simulation" unit="1/time"/>
  <parameters name="k_example_DFE" type="CONSTANT" expression="0.6" description="Example value used in a DFE global stability simulation (constant k)" unit="1/time"/>
  <parameters name="σ_example_DFE" type="CONSTANT" expression="0.007" description="Example value used in a DFE global stability simulation" unit="dimensionless"/>
  <parameters name="βₕ_example_DFE" type="CONSTANT" expression="18" description="Example value used in a DFE global stability simulation (constant βₕ)" unit="1/time"/>
  <parameters name="γ₁_example_DFE" type="CONSTANT" expression="0.5" description="Example value used in a DFE global stability simulation (constant γ₁)" unit="1/time"/>
  <parameters name="γ₂_example_DFE" type="CONSTANT" expression="0.5" description="Example value used in a DFE global stability simulation (constant γ₂)" unit="1/time"/>
  <parameters name="ρ_example_DFE" type="CONSTANT" expression="0.5" description="Example value used in a DFE global stability simulation" unit="dimensionless"/>
  <parameters name="μᵥ_example_DFE" type="CONSTANT" expression="0.65" description="Example value used in a DFE global stability simulation" unit="1/time"/>

  <parameters name="B_example_endemic" type="CONSTANT" expression="200" description="Example value used in an endemic simulation" unit="hosts/time"/>

  <parameters name="k_example_endemic" type="CONSTANT" expression="0.6" description="Example value used in an endemic simulation (constant k)" unit="1/time"/>

  <parameters name="βₕ_example_endemic" type="CONSTANT" expression="50" description="Example value used in an endemic simulation (constant βₕ)" unit="1/time"/>
  <parameters name="γ₁_example_endemic" type="CONSTANT" expression="0.5" description="Example value used in an endemic simulation (constant γ₁)" unit="1/time"/>
  <parameters name="γ₂_example_endemic" type="CONSTANT" expression="0.5" description="Example value used in an endemic simulation (constant γ₂)" unit="1/time"/>
  <parameters name="ρ_example_endemic" type="CONSTANT" expression="0.5" description="Example value used in an endemic simulation" unit="dimensionless"/>
  <parameters name="μᵥ_example_endemic" type="CONSTANT" expression="0.2" description="Example value used in an endemic simulation" unit="1/time"/>
  <parameters name="βᵥ_example_endemic" type="CONSTANT" expression="30" description="Example value used in an endemic simulation (constant βᵥ)" unit="1/time"/>

  <!-- ===================== -->
  <!-- Compartments          -->
  <!-- ===================== -->
  <compartments PrimaryName="Susceptible" SecondaryName="Human (age-structured) S_h(a,t)" population="0.0">
    <!-- S_h -> E_h: β_h(a) * I_v(t) / N_h(t) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.5"
                   contactCompartment="//@compartments.5"
                   target="//@compartments.1"
                   description="Primary infection of susceptible humans by infectious mosquitoes (standard incidence βₕ(a) Iᵥ / Nₕ)."/>
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="Human (age-structured) E_h(a,t)" population="0.0">
    <!-- E_h -> I_h: ρ k(a) -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.8"
                   target="//@compartments.2"
                   description="Progression from exposed to symptomatic infectious humans at rate ρ k(a)."/>
    <!-- E_h -> A_h: (1-ρ) k(a) -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.8"
                   target="//@compartments.3"
                   description="Progression from exposed to asymptomatic infectious humans at rate (1-ρ) k(a). (ρ handled in equation-level split)."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Human symptomatic (age-structured) I_h(a,t)" population="0.0">
    <!-- I_h -> Recovered: γ1(a) -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.10"
                   target="//@compartments.4"
                   description="Recovery of symptomatic infectious humans at rate γ₁(a)."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Human asymptomatic (age-structured) A_h(a,t)" population="0.0">
    <!-- A_h -> I_h: σ β_h(a) * I_v(t) / N_h(t)  (reinfection/secondary infection leading to symptomatic) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.5"
                   contactCompartment="//@compartments.5"
                   target="//@compartments.2"
                   description="Secondary infection (reinfection) of asymptomatic humans leading to symptomatic infection; contact term σ βₕ(a) Iᵥ / Nₕ(t). (σ applied in equation-level term)."/>
    <!-- A_h -> Recovered: γ2(a) -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.11"
                   target="//@compartments.4"
                   description="Recovery of asymptomatic infectious humans at rate γ₂(a)."/>
  </compartments>

  <compartments PrimaryName="Recovered" SecondaryName="Human (age-structured) R_h(a,t)" population="0.0"/>

  <compartments PrimaryName="Infectious Mosquitoes" SecondaryName="Vector I_v(t)" population="0.0"/>

  <!-- ===================== -->
  <!-- Demography: Births     -->
  <!-- ===================== -->
  <birthSources name="HumanRecruitment"
                rateParameter="//@parameters.2"
                targetCompartment="//@compartments.0"
                fixedRate="true"/>

  <!-- ===================== -->
  <!-- Death sinks (natural)  -->
  <!-- ===================== -->
  <deathSinks name="Natural_Death_S_h" rateParameter="//@parameters.3" sourceCompartment="//@compartments.0"/>
  <deathSinks name="Natural_Death_E_h" rateParameter="//@parameters.3" sourceCompartment="//@compartments.1"/>
  <deathSinks name="Natural_Death_I_h" rateParameter="//@parameters.3" sourceCompartment="//@compartments.2"/>
  <deathSinks name="Natural_Death_A_h" rateParameter="//@parameters.3" sourceCompartment="//@compartments.3"/>
  <deathSinks name="Natural_Death_R_h" rateParameter="//@parameters.3" sourceCompartment="//@compartments.4"/>
  <deathSinks name="Natural_Death_I_v" rateParameter="//@parameters.4" sourceCompartment="//@compartments.5"/>

  <!-- ===================================================== -->
  <!-- Notes on vector infection term (not representable as a -->
  <!-- simple ContactFlow in this metamodel without S_v):     -->
  <!--                                                      -->
  <!-- Paper equation:                                       -->
  <!-- dI_v/dt = (1 - I_v) * (1/N_h) ∫ β_v(a) (I_h + σ A_h) da -->
  <!--          - μ_v I_v                                    -->
  <!-- This is logistic-type infection of vectors with        -->
  <!-- susceptible fraction (1 - I_v). The figure shows an SI -->
  <!-- vector system with S_v -> I_v driven by infectious     -->
  <!-- humans. To encode exactly, one would include S_v(t) as -->
  <!-- a compartment and a ContactFlow S_v -> I_v with a      -->
  <!-- composite force of infection from I_h and A_h.         -->
  <!-- Here we keep I_v as the modeled state (as in equations)-->
  <!-- and represent mosquito infection as an equation-level  -->
  <!-- EXPRESSION parameter (∫...) rather than a metamodel    -->
  <!-- flow.                                                 -->
  <!-- ===================================================== -->

</seir:SEIRModel>
