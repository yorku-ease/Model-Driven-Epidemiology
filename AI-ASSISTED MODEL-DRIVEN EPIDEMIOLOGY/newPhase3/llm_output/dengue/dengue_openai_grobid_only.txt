```xml
<?xml version="1.0" encoding="UTF-8"?>
<seir:SEIRModel xmi:version="2.0"
    xmlns:xmi="http://www.omg.org/XMI"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xmlns:seir="http://example.com/seirmodel">

  <!-- ===================== -->
  <!-- Parameters (symbolic) -->
  <!-- ===================== -->
  <parameters name="a" type="VARIABLE" description="Host chronological age (structuring variable a in densities Sh(a,t), Eh(a,t), Ih(a,t), Ah(a,t), Rh(a,t))" unit="time"/>
  <parameters name="t" type="VARIABLE" description="Time" unit="time"/>

  <parameters name="B" type="CONSTANT" description="Influx (recruitment) rate of susceptible human population" unit="humans/time"/>
  <parameters name="μₕ" type="CONSTANT" description="Natural death rate of human hosts" unit="1/time"/>

  <parameters name="βₕ(a)" type="VARIABLE" description="Age-dependent transmission/contact rate from infectious mosquitoes to susceptible humans (standard incidence term uses βₕ(a) Iᵥ Sh / Nh)" unit="1/time"/>
  <parameters name="k(a)" type="VARIABLE" description="Age-dependent progression rate from exposed humans to infectious humans (end of latent period)" unit="1/time"/>
  <parameters name="ρ" type="CONSTANT" description="Probability an exposed human becomes symptomatic infectious (0&lt;ρ&lt;1); fraction (1-ρ) becomes asymptomatic infectious" unit="dimensionless"/>
  <parameters name="γ₁(a)" type="VARIABLE" description="Age-specific recovery rate of symptomatic infectious human hosts" unit="1/time"/>
  <parameters name="γ₂(a)" type="VARIABLE" description="Age-specific recovery rate of asymptomatic infectious human hosts" unit="1/time"/>
  <parameters name="σ" type="CONSTANT" description="Reinfection intensity factor: scales contact rate for reinfection of asymptomatic humans (σ βₕ(a))" unit="dimensionless"/>

  <parameters name="μᵥ" type="CONSTANT" description="Natural death rate of mosquitoes" unit="1/time"/>
  <parameters name="βᵥ(a)" type="VARIABLE" description="Age-specific transmission rate from infectious human hosts to susceptible mosquitoes" unit="1/time"/>

  <!-- Normalization uses Nh(t) and P∞(a); kept as symbolic parameters/functions since explicit formulas not provided in extracted text -->
  <parameters name="Nₕ(t)" type="VARIABLE" description="Total number of humans at time t (used in standard incidence terms)" unit="humans"/>
  <parameters name="P∞(a)" type="VARIABLE" description="Limiting age distribution used for normalization (Sh=sh P∞, etc.)" unit="1/age"/>

  <!-- ===================== -->
  <!-- Compartments (states) -->
  <!-- ===================== -->
  <!-- Humans (age-structured densities; represented as stratified by 'Age a' in the paper) -->
  <compartments PrimaryName="Susceptible" SecondaryName="Humans (age a)" population="0.0">
    <!-- Infection by infectious mosquitoes: Sh -> Eh -->
    <outgoingFlows xsi:type="seir:ContactFlow"
        contactRateParameter="//@parameters.3"
        contactCompartment="//@compartments.5"
        target="//@compartments.1"
        description="Susceptible humans become exposed via bites from infectious mosquitoes (standard incidence uses βₕ(a) Iᵥ Sh / Nₕ)."/>
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="Humans (age a)" population="0.0">
    <!-- Progression: Eh -> symptomatic Ih with fraction ρ, and Eh -> asymptomatic Ah with fraction (1-ρ).
         Metamodel RateFlow cannot encode fractions directly; represent as two RateFlows with expression rates. -->
    <outgoingFlows xsi:type="seir:RateFlow"
        rateParameter="//@parameters.4"
        target="//@compartments.2"
        description="Progression from exposed to symptomatic infectious humans at rate ρ·k(a) (paper: fraction ρ becomes symptomatic)."/>
    <outgoingFlows xsi:type="seir:RateFlow"
        rateParameter="//@parameters.4"
        target="//@compartments.3"
        description="Progression from exposed to asymptomatic infectious humans at rate (1-ρ)·k(a) (paper: fraction 1-ρ becomes asymptomatic)."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Humans (Symptomatic, age a)" population="0.0">
    <outgoingFlows xsi:type="seir:RateFlow"
        rateParameter="//@parameters.6"
        target="//@compartments.4"
        description="Recovery of symptomatic infectious humans at age-specific rate γ₁(a)."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Humans (Asymptomatic, age a)" population="0.0">
    <!-- Recovery -->
    <outgoingFlows xsi:type="seir:RateFlow"
        rateParameter="//@parameters.7"
        target="//@compartments.4"
        description="Recovery of asymptomatic infectious humans at age-specific rate γ₂(a)."/>
    <!-- Reinfection: asymptomatic -> symptomatic via infectious mosquitoes with contact rate σ βₕ(a) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
        contactRateParameter="//@parameters.3"
        contactCompartment="//@compartments.5"
        target="//@compartments.2"
        description="Reinfection of asymptomatic infectious humans leading to symptomatic infection; contact rate σ·βₕ(a) with infectious mosquitoes (σ distinguishes primary vs secondary infection intensity)."/>
  </compartments>

  <compartments PrimaryName="Recovered" SecondaryName="Humans (age a)" population="0.0">
    <!-- Paper notes Rh equation is decoupled; no outgoing flows specified in extracted sections -->
  </compartments>

  <!-- Mosquitoes (unstructured by age in the extracted text) -->
  <compartments PrimaryName="Susceptible Mosquitoes" population="0.0">
    <!-- Infection from infectious humans (symptomatic and asymptomatic) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
        contactRateParameter="//@parameters.10"
        contactCompartment="//@compartments.2"
        target="//@compartments.5"
        description="Susceptible mosquitoes become infectious via contact with symptomatic infectious humans (rate uses βᵥ(a) in paper)."/>
    <outgoingFlows xsi:type="seir:ContactFlow"
        contactRateParameter="//@parameters.10"
        contactCompartment="//@compartments.3"
        target="//@compartments.5"
        description="Susceptible mosquitoes become infectious via contact with asymptomatic infectious humans (rate uses βᵥ(a) in paper)."/>
  </compartments>

  <compartments PrimaryName="Infectious Mosquitoes" population="0.0">
    <!-- No recovery in mosquitoes in extracted text -->
  </compartments>

  <!-- ===================== -->
  <!-- Birth / recruitment    -->
  <!-- ===================== -->
  <birthSources name="HumanRecruitment"
      rateParameter="//@parameters.2"
      targetCompartment="//@compartments.0"
      fixedRate="true"/>

  <!-- ===================== -->
  <!-- Death sinks (natural)  -->
  <!-- ===================== -->
  <!-- Humans: natural death from each human compartment at rate μₕ -->
  <deathSinks name="NaturalDeath_SusceptibleHumans" rateParameter="//@parameters.3" sourceCompartment="//@compartments.0"/>
  <deathSinks name="NaturalDeath_ExposedHumans" rateParameter="//@parameters.3" sourceCompartment="//@compartments.1"/>
  <deathSinks name="NaturalDeath_SymptomaticHumans" rateParameter="//@parameters.3" sourceCompartment="//@compartments.2"/>
  <deathSinks name="NaturalDeath_AsymptomaticHumans" rateParameter="//@parameters.3" sourceCompartment="//@compartments.3"/>
  <deathSinks name="NaturalDeath_RecoveredHumans" rateParameter="//@parameters.3" sourceCompartment="//@compartments.4"/>

  <!-- Mosquitoes: natural death from each mosquito compartment at rate μᵥ -->
  <deathSinks name="NaturalDeath_SusceptibleMosquitoes" rateParameter="//@parameters.9" sourceCompartment="//@compartments.6"/>
  <deathSinks name="NaturalDeath_InfectiousMosquitoes" rateParameter="//@parameters.9" sourceCompartment="//@compartments.7"/>

</seir:SEIRModel>
```
