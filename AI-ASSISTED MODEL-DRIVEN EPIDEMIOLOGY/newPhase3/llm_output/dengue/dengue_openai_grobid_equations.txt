  <!-- ===================== -->
  <parameters name="a" type="VARIABLE" description="Chronological age of human host (structuring variable in PDE system)" unit="time"/>
  <parameters name="t" type="VARIABLE" description="Time" unit="time"/>

  <parameters name="βᵥ(a)" type="VARIABLE"

  <parameters name="ρ" type="CONSTANT"

              unit="dimensionless"/>

  <parameters name="σ" type="CONSTANT"
              description="Reinfection intensity factor for asymptomatic humans; scales mosquito-to-human contact rate for secondary infection of asymptomatic hosts and also weights asymptomatic contribution to mosquito infection in the vector equation"
              unit="dimensionless"/>

  <parameters name="γ₁(a)" type="VARIABLE"
              description="Age-specific recovery rate of symptomatic infectious humans"
              unit="1/time"/>
  <parameters name="γ₂(a)" type="VARIABLE"
              description="Age-specific recovery rate of asymptomatic infectious humans"
              unit="1/time"/>

  <parameters name="Nₕ(t)" type="EXPRESSION"
              description="Total human population at time t (normalization in standard incidence terms)"
              unit="persons"/>

  <!-- Mentioned in text (influx), but not present in the provided equations block -->
  <parameters name="B" type="CONSTANT"
              description="Constant influx/recruitment rate into susceptible human population (mentioned in text; boundary condition at age 0 in PDE formulation)"
              unit="persons/time"/>

  <!-- ===================== -->
  <!-- Compartments          -->
  <!-- ===================== -->
  <!-- Note: Human compartments are age-structured densities (a,t). Vector compartment is time-only. -->
  <compartments PrimaryName="Susceptible" SecondaryName="Humans (age-structured)" population="0.0">
    <!-- Outflows: infection by infectious mosquitoes (standard incidence), and natural death -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.2"
                   contactCompartment="//@compartments.5"
                   target="//@compartments.1"
                   description="Primary infection of susceptible humans by infectious mosquitoes: βₕ(a) Iᵥ(t) Sₕ(a,t) / Nₕ(t)"/>
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.4"
                   target="//@compartments.6"
                   description="Natural death of susceptible humans: μₕ Sₕ(a,t)"/>
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="Humans (age-structured)" population="0.0">
    <!-- Inflow: from susceptible via mosquito bites -->

    <outgoingFlows xsi:type="seir:RateFlow"

    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.6"
                   target="//@compartments.3"
                   description="Progression from exposed to asymptomatic infectious humans (base rate k(a); fraction (1-ρ) applied in equation term (1-ρ) k(a) Eₕ)"/>
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.4"
                   target="//@compartments.6"
                   description="Natural death of exposed humans: μₕ Eₕ(a,t)"/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Humans (symptomatic, age-structured)" population="0.0">
    <!-- Inflows: from exposed (symptomatic fraction), and from asymptomatic via reinfection becoming symptomatic -->
    <!-- Outflows: recovery and natural death -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.9"
                   target="//@compartments.4"
                   description="Recovery of symptomatic infectious humans: γ₁(a) Iₕ(a,t)"/>
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.4"
                   target="//@compartments.6"
                   description="Natural death of symptomatic infectious humans: μₕ Iₕ(a,t)"/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Humans (asymptomatic, age-structured)" population="0.0">
    <!-- Inflow: from exposed (asymptomatic fraction) -->
    <!-- Outflows: reinfection to symptomatic (contact with infectious mosquitoes), recovery, and natural death -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.2"
                   contactCompartment="//@compartments.5"
                   target="//@compartments.2"
                   description="Secondary infection of asymptomatic humans by infectious mosquitoes leading to symptomatic infection: σ βₕ(a) Iᵥ(t) Aₕ(a,t) / Nₕ(t)"/>
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.10"
                   target="//@compartments.4"
                   description="Recovery of asymptomatic infectious humans: γ₂(a) Aₕ(a,t)"/>
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.4"
                   target="//@compartments.6"
                   description="Natural death of asymptomatic infectious humans: μₕ Aₕ(a,t)"/>
  </compartments>

  <compartments PrimaryName="Recovered" SecondaryName="Humans (age-structured)" population="0.0">
    <!-- Outflow: natural death (equation includes -μₕ Rₕ) -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.4"
                   target="//@compartments.6"
                   description="Natural death of recovered humans: μₕ Rₕ(a,t)"/>
  </compartments>

  <compartments PrimaryName="Infectious Mosquitoes" SecondaryName="Vectors" population="0.0">
    <!-- Outflow: natural death -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.5"
                   target="//@compartments.6"
                   description="Natural death of infectious mosquitoes: μᵥ Iᵥ(t)"/>
  </compartments>

  <compartments PrimaryName="Death" SecondaryName="Natural (humans and mosquitoes)" population="0.0"/>

  <!-- ===================== -->
  <!-- Mentioned in text: constant influx B into susceptible humans (implemented as BirthSource). -->
  <birthSources name="Human recruitment (age 0 boundary inflow)"

                fixedRate="true"/>

  <!-- Human natural deaths from each human compartment -->
  <deathSinks name="Natural death - Susceptible humans" rateParameter="//@parameters.4" sourceCompartment="//@compartments.0"/>
  <deathSinks name="Natural death - Exposed humans"     rateParameter="//@parameters.4" sourceCompartment="//@compartments.1"/>
  <deathSinks name="Natural death - Symptomatic humans" rateParameter="//@parameters.4" sourceCompartment="//@compartments.2"/>
  <deathSinks name="Natural death - Asymptomatic humans" rateParameter="//@parameters.4" sourceCompartment="//@compartments.3"/>
  <deathSinks name="Natural death - Recovered humans"   rateParameter="//@parameters.4" sourceCompartment="//@compartments.4"/>

  <!-- Mosquito natural death -->
  <deathSinks name="Natural death - Infectious mosquitoes" rateParameter="//@parameters.5" sourceCompartment="//@compartments.5"/>

  <!-- ===================== -->
  <!-- Notes on missing pieces -->
  <!-- ===================== -->
  <!--

    Contact terms involving division by Nₕ(t) are represented as ContactFlows with βₕ(a) as the contact rate parameter;
