<?xml version="1.0" encoding="UTF-8"?>
<seir:SEIRModel
    xmi:version="2.0"
    xmlns:xmi="http://www.omg.org/XMI"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xmlns:seir="http://example.com/seirmodel">

  <!-- ===================== -->
  <!-- Parameters (symbolic) -->
  <!-- ===================== -->
  <parameters name="λ(t)" type="EXPRESSION"
              expression="β * (I₁(t) + I₂(t))"
              description="Force of infection at time t; proportional to infectious individuals under homogeneous mixing."
              unit="1/day"/>

  <parameters name="β" type="CONSTANT"
              description="Transmission rate (rate at which new infections take place given a contact); used in λ(t)."
              unit="1/day"/>

  <parameters name="σ" type="CONSTANT"
              description="Daily progression rate through exposed/latent stages (E₁→E₂ and E₂→I₁). Mean latent/incubation period d_L = 2*(1/σ) days."
              unit="1/day"/>

  <parameters name="γ" type="CONSTANT"
              description="Daily progression rate through infectious stages (I₁→I₂ and I₂→R). Mean infectious period d_I = 2*(1/γ) days."
              unit="1/day"/>

  <!-- Observational-model parameters mentioned in text/figure (not part of transmission ODE system) -->
  <parameters name="p_ICU" type="CONSTANT"
              description="Probability (proportion) of infected individuals who are admitted to ICU/HDU (severity given infection)."
              unit="dimensionless"/>

  <parameters name="μ_ICU|E" type="CONSTANT"
              description="Mean delay from infection/exposure to ICU/HDU admission (used in convolution observational model; Gamma-distributed waiting time)."
              unit="day"/>

  <parameters name="σ²_ICU|E" type="CONSTANT"
              description="Variance of delay from infection/exposure to ICU/HDU admission (Gamma waiting time)."
              unit="day^2"/>

  <parameters name="η" type="CONSTANT"
              description="Negative Binomial dispersion parameter for weekly ICU/HDU admission counts."
              unit="dimensionless"/>

  <!-- Parameters referenced in text for inference/re-parameterization (not directly in the provided state-update equations) -->
  <parameters name="π" type="CONSTANT"
              description="Initial susceptibility / proportion non-immune at start of season (used for identifiability and to compute effective reproduction number R_n)."
              unit="dimensionless"/>

  <parameters name="λ₀" type="CONSTANT"
              description="Initial infection rate / initial force of infection (used in early exponential growth re-parameterization)."
              unit="1/day"/>

  <parameters name="ψ" type="CONSTANT"
              description="Early exponential growth rate (EGR) parameter used to re-parameterize the transmission model at epidemic start."
              unit="1/day"/>

  <!-- ===================== -->
  <!-- Compartments (states) -->
  <!-- ===================== -->
  <compartments PrimaryName="Susceptible" SecondaryName="S(t)"
                description="Susceptible individuals in the population at time t."
                population="0">
    <!-- S -> E1 via contact with infectious individuals (λ(t) = β*(I1+I2)) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.1"
                   contactCompartment="//@compartments.3"
                   target="//@compartments.1"
                   description="Infection: Susceptible become exposed (enter E₁) at rate λ(t)S, with λ(t) proportional to infectious individuals."/>
    <!-- Note: contactCompartment points to I₁; λ(t) expression includes I₁+I₂ at parameter level. -->
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="E₁(t)"
                description="First exposed/latent stage (infected but not yet infectious); part of Erlang-2 latent period."
                population="0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.2"
                   target="//@compartments.2"
                   description="Progression through latent period: E₁ → E₂ at rate σ."/>
    <!-- ICU linkage is observational (delay + p_ICU), not a transmission-state flow in the provided equations -->
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="E₂(t)"
                description="Second exposed/latent stage; completes Erlang-2 latent period before becoming infectious."
                population="0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.2"
                   target="//@compartments.3"
                   description="Onset of infectiousness: E₂ → I₁ at rate σ."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="I₁(t)"
                description="First infectious stage; part of Erlang-2 infectious period."
                population="0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.3"
                   target="//@compartments.4"
                   description="Progression within infectious period: I₁ → I₂ at rate γ."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="I₂(t)"
                description="Second infectious stage; individuals recover/are removed from here."
                population="0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.3"
                   target="//@compartments.5"
                   description="Recovery/removal: I₂ → R at rate γ."/>
  </compartments>

  <compartments PrimaryName="Recovered" SecondaryName="R(t)"
                description="Removed/Recovered individuals (no longer infectious; assumed immune)."
                population="0"/>

  <!-- ===================== -->
  <!-- Notes on ICU/HDU data linkage (observational model) -->
  <!-- ===================== -->
  <!--
    The figure shows a curved arrow from E₁(t) to an ICU_t box with labels p_ICU, μ_ICU|E, σ_ICU|E.
    The text specifies ICU/HDU admissions are generated by:
      - a fraction p_ICU of infections
      - after a Gamma-distributed delay from infection to ICU admission (mean μ_ICU|E, variance σ²_ICU|E)
      - weekly observed counts ICU_w ~ NegBin(mean μ_w, dispersion η)
    This is an observation process (convolution over past infections), not a compartmental state transition in the provided SEIR equations.
    Therefore ICU_t is not encoded as a disease-state compartment/flow here, but its parameters are included above.
  -->

</seir:SEIRModel>
