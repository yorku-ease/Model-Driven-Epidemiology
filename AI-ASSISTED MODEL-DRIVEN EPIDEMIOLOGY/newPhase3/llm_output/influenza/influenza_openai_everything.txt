```xml
<?xml version="1.0" encoding="UTF-8"?>
<seir:SEIRModel
    xmi:version="2.0"
    xmlns:xmi="http://www.omg.org/XMI"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xmlns:seir="http://example.com/seirmodel"
>
  <!--
    Model extracted from: Corbella et al. (2017) "Monitoring and predicting influenza epidemics from routinely collected severe case data"
    Priority used: equations > tables > sections > full text > figures.

    Notes:
    - Core transmission model is deterministic, discrete-time SEIR with 2-stage E and 2-stage I (E1,E2,I1,I2).
    - Force of infection: λ(t) = β * I(t) / N, where I(t) = I1(t)+I2(t).
    - Observations (ICU_w) are modeled via a convolution from infections with delay ~ Gamma(mean μ_ICU|E, var σ^2_ICU|E),
      and weekly counts are NegBin(mean μ_w, dispersion η). This observational layer is not representable as compartments/flows
      in the provided SEIR metamodel, so parameters are included but no ICU compartment/flows are instantiated.
  -->

  <parameters name="δt" type="CONSTANT" expression="0.25" description="Discrete time step length (quarter of a day)" unit="day"/>
  <parameters name="N" type="VARIABLE" description="Total population size for the season (England)" unit="persons"/>
  <parameters name="β" type="CONSTANT" description="Transmission rate parameter in λ(t)=β*I(t)/N (frequency-dependent)" unit="1/day"/>

  <parameters name="σ" type="CONSTANT" expression="1" description="Progression rate through exposed stages (E1→E2 and E2→I1); implies mean latent period d_L=2/σ" unit="1/day"/>
  <parameters name="γ" type="CONSTANT" expression="0.5797" description="Progression/recovery rate through infectious stages (I1→I2 and I2→R); implies mean infectious period d_I=2/γ" unit="1/day"/>

  <!-- EGR reparameterisation parameters used for inference (appear in text/tables) -->
  <parameters name="π" type="VARIABLE" description="Initial susceptibility: proportion of population initially susceptible" unit="dimensionless"/>
  <parameters name="λ₀" type="VARIABLE" description="Initial infection rate at t=0 under EGR parametrisation (used for inference)" unit="1/day"/>
  <parameters name="ψ" type="VARIABLE" description="Exponential growth rate (EGR) at epidemic start (used for inference)" unit="1/day"/>

  <!-- Observation model parameters (ICU admissions) -->
  <parameters name="p_ICU" type="VARIABLE" description="Probability of ICU/HDU admission given infection" unit="dimensionless"/>
  <parameters name="μ_ICU|E" type="CONSTANT" expression="5.708" description="Mean delay from infection to ICU/HDU admission (Gamma approximation)" unit="day"/>
  <parameters name="σ²_ICU|E" type="CONSTANT" expression="18.24" description="Variance of delay from infection to ICU/HDU admission (Gamma approximation)" unit="day^2"/>
  <parameters name="η" type="VARIABLE" description="Negative binomial dispersion parameter for weekly ICU/HDU counts" unit="dimensionless"/>

  <!-- Derived/mentioned quantities (no fixed values in main table; included for completeness) -->
  <parameters name="λ(t)" type="EXPRESSION" expression="β * (I1 + I2) / N" description="Force of infection at time t" unit="1/day"/>
  <parameters name="R₀" type="VARIABLE" description="Basic reproduction number (defined via EGR parametrisation in paper)" unit="dimensionless"/>
  <parameters name="Rₙ" type="VARIABLE" description="Effective reproduction number in partially susceptible population" unit="dimensionless"/>
  <parameters name="TAR" type="VARIABLE" description="Total attack rate over the season (probability a susceptible individual is infected)" unit="dimensionless"/>
  <parameters name="I₀" type="VARIABLE" description="Initial total infectious individuals (derived quantity reported in results)" unit="persons"/>

  <!-- Compartments (core transmission model) -->
  <compartments PrimaryName="Susceptible" SecondaryName="S(t)" population="0" />
  <compartments PrimaryName="Exposed" SecondaryName="E1(t)" population="0" />
  <compartments PrimaryName="Exposed" SecondaryName="E2(t)" population="0" />
  <compartments PrimaryName="Infectious" SecondaryName="I1(t)" population="0" />
  <compartments PrimaryName="Infectious" SecondaryName="I2(t)" population="0" />
  <compartments PrimaryName="Removed" SecondaryName="R(t)" population="0" />

  <!-- Flows (from discrete-time equations; represented as continuous-time rates consistent with metamodel) -->

  <!-- S -> E1 : infection (contact-based), λ(t) = β*(I1+I2)/N -->
  <compartments PrimaryName="Susceptible" SecondaryName="S(t)">
    <outgoingFlows
        xsi:type="seir:ContactFlow"
        contactRateParameter="//@parameters.2"
        contactCompartment="//@compartments.3"
        target="//@compartments.1"
        description="Infection: S→E1 with force of infection λ(t)=β*(I1+I2)/N. Implemented as contact with I1 (see note)."
    />
    <!--
      NOTE: The metamodel ContactFlow allows only one contactCompartment.
      The paper uses I(t)=I1+I2. To remain faithful, β is provided and λ(t) is documented as an EXPRESSION parameter.
      A simulator should implement λ(t) using both I1 and I2.
    -->
  </compartments>

  <!-- E1 -> E2 -->
  <compartments PrimaryName="Exposed" SecondaryName="E1(t)">
    <outgoingFlows
        xsi:type="seir:RateFlow"
        rateParameter="//@parameters.3"
        target="//@compartments.2"
        description="Latent progression: E1→E2 at rate σ"
    />
  </compartments>

  <!-- E2 -> I1 -->
  <compartments PrimaryName="Exposed" SecondaryName="E2(t)">
    <outgoingFlows
        xsi:type="seir:RateFlow"
        rateParameter="//@parameters.3"
        target="//@compartments.3"
        description="Latent progression: E2→I1 at rate σ"
    />
  </compartments>

  <!-- I1 -> I2 -->
  <compartments PrimaryName="Infectious" SecondaryName="I1(t)">
    <outgoingFlows
        xsi:type="seir:RateFlow"
        rateParameter="//@parameters.4"
        target="//@compartments.4"
        description="Infectious progression: I1→I2 at rate γ"
    />
  </compartments>

  <!-- I2 -> R -->
  <compartments PrimaryName="Infectious" SecondaryName="I2(t)">
    <outgoingFlows
        xsi:type="seir:RateFlow"
        rateParameter="//@parameters.4"
        target="//@compartments.5"
        description="Recovery/removal: I2→R at rate γ"
    />
  </compartments>

</seir:SEIRModel>
```
