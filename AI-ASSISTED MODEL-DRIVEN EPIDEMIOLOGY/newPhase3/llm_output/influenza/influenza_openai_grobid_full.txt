  <!-- =========================================================
       Parameters (paper symbols; values from extracted tables)
       Notes:
       - Core transmission equations are discrete-time with step δt = 0.25 day.
       - Force of infection: λ(t) = β * I(t) / N, where I(t)=I1(t)+I2(t).
       - Some parameters (π, λ0, ψ, η, pICU) are inferred with priors (Table 1).
       - σ, γ, μICU|E, σ^2_ICU|E are treated as known (Table 1).
       ========================================================= -->
  <parameters name="δt" type="CONSTANT" expression="0.25" description="Discrete time step length" unit="day"/>

  <parameters name="N" type="VARIABLE" description="Population size for the season (use N2012/13, N2013/14, or N2014/15)" unit="persons"/>
  <parameters name="N2012/13" type="CONSTANT" expression="53679750" description="Population size, season 2012/13" unit="persons"/>
  <parameters name="N2013/14" type="CONSTANT" expression="54091200" description="Population size, season 2013/14" unit="persons"/>
  <parameters name="N2014/15" type="CONSTANT" expression="54551450" description="Population size, season 2014/15" unit="persons"/>

  <parameters name="β" type="VARIABLE" description="Transmission rate (rate at which new infections take place given a contact); used in λ(t)=β*I/N" unit="per day"/>
  <parameters name="λ(t)" type="EXPRESSION" expression="β * (I1 + I2) / N" description="Force of infection at time t" unit="per day"/>

  <parameters name="σ" type="CONSTANT" expression="1" description="Daily rate of progression through latent stages (E1→E2 and E2→I1); mean latent/incubation period dL=2/σ" unit="per day"/>
  <parameters name="γ" type="CONSTANT" expression="0.5797" description="Daily rate of progression through infectious stages (I1→I2 and I2→R); mean infectious period dI=2/γ" unit="per day"/>

  <!-- Inference parameters / priors (Table 1: flat priors) -->
  <parameters name="π" type="VARIABLE" description="Initial susceptibility (proportion susceptible at start of season)" unit="dimensionless"/>
  <parameters name="λ0" type="VARIABLE" description="Initial infection rate parameter used in early growth re-parameterisation (EGR)" unit="per day"/>
  <parameters name="ψ" type="VARIABLE" description="Exponential growth rate parameter used in early growth re-parameterisation (EGR)" unit="per day"/>
  <parameters name="η" type="VARIABLE" description="Negative binomial over-dispersion parameter for weekly ICU counts" unit="dimensionless"/>
  <parameters name="pICU" type="VARIABLE" description="Probability of ICU/HDU admission given infection" unit="dimensionless"/>

  <!-- Prior bounds as parameters (to preserve all information from Table 1) -->
  <parameters name="π_prior_low" type="CONSTANT" expression="0" description="Lower bound of prior for π ~ Unif(0,1)" unit="dimensionless"/>
  <parameters name="π_prior_high" type="CONSTANT" expression="1" description="Upper bound of prior for π ~ Unif(0,1)" unit="dimensionless"/>

  <parameters name="λ0_prior_low" type="CONSTANT" expression="0" description="Lower bound of prior for λ0 ~ Unif(0,0.000241)" unit="per day"/>
  <parameters name="λ0_prior_high" type="CONSTANT" expression="0.000241" description="Upper bound of prior for λ0 ~ Unif(0,0.000241)" unit="per day"/>

  <parameters name="ψ_prior_low" type="CONSTANT" expression="0" description="Lower bound of prior for ψ ~ Unif(0,0.39)" unit="per day"/>
  <parameters name="ψ_prior_high" type="CONSTANT" expression="0.39" description="Upper bound of prior for ψ ~ Unif(0,0.39)" unit="per day"/>

  <parameters name="η_prior_low" type="CONSTANT" expression="1" description="Lower bound of prior for η ~ Unif(1,100)" unit="dimensionless"/>
  <parameters name="η_prior_high" type="CONSTANT" expression="100" description="Upper bound of prior for η ~ Unif(1,100)" unit="dimensionless"/>

  <parameters name="pICU_prior_low" type="CONSTANT" expression="0" description="Lower bound of prior for pICU ~ Unif(0,1)" unit="dimensionless"/>
  <parameters name="pICU_prior_high" type="CONSTANT" expression="1" description="Upper bound of prior for pICU ~ Unif(0,1)" unit="dimensionless"/>

  <!-- Observational delay distribution parameters (treated as known; Table 1) -->
  <parameters name="μICU|E" type="CONSTANT" expression="5.708" description="Mean time from infection to ICU/HDU admission (delay distribution mean)" unit="day"/>
  <parameters name="σ2ICU|E" type="CONSTANT" expression="18.24" description="Variance of time from infection to ICU/HDU admission (delay distribution variance)" unit="day^2"/>

  <!-- Derived/mentioned epidemiological quantities (appear in results table; kept as parameters for completeness) -->
  <parameters name="R0" type="VARIABLE" description="Basic reproduction number (derived from β, σ, γ and initial conditions; reported in results)" unit="dimensionless"/>
  <parameters name="Rn" type="VARIABLE" description="Effective reproduction number in partially immune population (function of π and ψ; reported in results)" unit="dimensionless"/>
  <parameters name="I0" type="VARIABLE" description="Initial total infectious individuals (derived; reported in results)" unit="persons"/>
  <parameters name="TAR" type="VARIABLE" description="Total attack rate over the season (probability of infection for a susceptible individual; reported in results)" unit="dimensionless"/>

  <!-- =========================================================
       Compartments (ground truth from equations; figure confirms)
       ========================================================= -->
  <compartments PrimaryName="Susceptible" SecondaryName="S" population="0">
    <!-- S -> E1: infection (contact with infectious) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
        contactRateParameter="//@parameters.5"
        contactCompartment="//@compartments.3"
        target="//@compartments.1"
        description="Infection of susceptibles at force of infection λ(t)=β*(I1+I2)/N (implemented as contact with I1; see note)."/>
    <!-- NOTE: Metamodel ContactFlow supports one contactCompartment; paper uses I=I1+I2.
         A second parallel ContactFlow from S to E1 with contactCompartment=I2 is added below. -->
    <outgoingFlows xsi:type="seir:ContactFlow"
        contactRateParameter="//@parameters.5"
        contactCompartment="//@compartments.4"
        target="//@compartments.1"
        description="Infection of susceptibles at force of infection λ(t)=β*(I1+I2)/N (second component for I2)."/>
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="E1" population="0">
    <!-- E1 -> E2 -->
    <outgoingFlows xsi:type="seir:RateFlow"
        rateParameter="//@parameters.6"
        target="//@compartments.2"
        description="Progression through latent period stage 1 (E1→E2) at rate σ."/>
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="E2" population="0">
    <!-- E2 -> I1 -->
    <outgoingFlows xsi:type="seir:RateFlow"
        rateParameter="//@parameters.6"
        target="//@compartments.3"
        description="Progression through latent period stage 2 (E2→I1) at rate σ."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="I1" population="0">
    <!-- I1 -> I2 -->
    <outgoingFlows xsi:type="seir:RateFlow"
        rateParameter="//@parameters.7"
        target="//@compartments.4"
        description="Progression through infectious period stage 1 (I1→I2) at rate γ."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="I2" population="0">
    <!-- I2 -> R -->
    <outgoingFlows xsi:type="seir:RateFlow"
        rateParameter="//@parameters.7"
        target="//@compartments.5"
        description="Recovery/removal (I2→R) at rate γ."/>
  </compartments>

  <compartments PrimaryName="Removed" SecondaryName="R" population="0"/>

  <!-- =========================================================
       ICU observation node (shown in Figure 2 and described in text)
       This is an auxiliary compartment-like observable; the paper links

       The metamodel has no explicit convolution/observation construct,
       so we include an ICU compartment with a documented incoming process.
       ========================================================= -->
  <compartments PrimaryName="ICU" SecondaryName="ICU_t" population="0">
    <!-- No mechanistic outflow specified in provided core equations -->
  </compartments>

  <!-- =========================================================
       Additional documented (non-state) process: ICU admissions generation
       Paper: μ_t = pICU * (infections in past) * GammaDelay(μICU|E, σ2ICU|E)
       Weekly observations: ICU_w ~ NegBin(μ_w, η)
       Not representable as standard SEIR flows without extra constructs.
       Included here as comments via parameters already defined.
       ========================================================= -->

</seir:SEIRModel>
