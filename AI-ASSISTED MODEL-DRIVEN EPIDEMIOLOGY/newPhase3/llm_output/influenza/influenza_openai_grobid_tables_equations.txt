```xml
<?xml version="1.0" encoding="UTF-8"?>
<seir:SEIRModel
    xmi:version="2.0"
    xmlns:xmi="http://www.omg.org/XMI"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xmlns:seir="http://example.com/seirmodel">

  <!-- ===================== -->
  <!-- Parameters (symbolic) -->
  <!-- ===================== -->
  <parameters name="δt" expression="0.25" type="CONSTANT"
              description="Discrete-time step length (quarter of a day) used in the transmission model update equations"
              unit="day"/>

  <parameters name="λ(t)" type="EXPRESSION"
              expression="β * (I₁ + I₂) / N"
              description="Force of infection at time t (per-susceptible infection hazard), frequency-dependent, proportional to infectious prevalence"
              unit="per day"/>

  <parameters name="β" type="CONSTANT"
              description="Transmission rate parameter (rate at which new infections take place given a contact); used in λ(t)=β I(t)/N"
              unit="per day"/>

  <parameters name="σ" expression="1" type="CONSTANT"
              description="Daily rate of progression through latent (exposed) stages: E₁→E₂ and E₂→I₁ (implies mean incubation period d_L = 2*(1/σ) days)"
              unit="per day"/>

  <parameters name="γ" expression="0.5797" type="CONSTANT"
              description="Daily rate of progression through infectious stages: I₁→I₂ and I₂→R (implies mean infectious period d_I = 2*(1/γ) days)"
              unit="per day"/>

  <parameters name="N" type="VARIABLE"
              description="Total population size for the season (fixed within season; use N_year value at simulation time)"
              unit="persons"/>

  <!-- Optional season-specific population sizes from Table 1 (provided as named constants for convenience) -->
  <parameters name="N2012/13" expression="53679750" type="CONSTANT"
              description="Population size used for season 2012/13"
              unit="persons"/>
  <parameters name="N2013/14" expression="54091200" type="CONSTANT"
              description="Population size used for season 2013/14"
              unit="persons"/>
  <parameters name="N2014/15" expression="54551450" type="CONSTANT"
              description="Population size used for season 2014/15"
              unit="persons"/>

  <!-- ===================== -->
  <!-- Compartments (states) -->
  <!-- ===================== -->
  <!-- Indices: 0..5 are referenced by flows below -->
  <compartments PrimaryName="Susceptible" SecondaryName="Influenza-susceptible population"
                population="0">
    <!-- Outflow: Susceptible -> Exposed stage 1 via infection (contact-based) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.2"
                   contactCompartment="//@compartments.3"
                   target="//@compartments.1"
                   description="New influenza infections: Susceptible individuals become Exposed (stage 1) at rate λ(t)=β*(I₁+I₂)/N"/>
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="Latent stage 1 (E₁)"
                population="0">
    <!-- Outflow: E1 -> E2 -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.3"
                   target="//@compartments.2"
                   description="Progression through incubation: Exposed stage 1 to Exposed stage 2 at rate σ"/>
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="Latent stage 2 (E₂)"
                population="0">
    <!-- Outflow: E2 -> I1 -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.3"
                   target="//@compartments.3"
                   description="End of incubation: Exposed stage 2 to Infectious stage 1 at rate σ"/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Infectious stage 1 (I₁)"
                population="0">
    <!-- Outflow: I1 -> I2 -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.4"
                   target="//@compartments.4"
                   description="Progression within infectious period: Infectious stage 1 to Infectious stage 2 at rate γ"/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="Infectious stage 2 (I₂)"
                population="0">
    <!-- Outflow: I2 -> Removed/Recovered -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.4"
                   target="//@compartments.5"
                   description="Recovery/removal: Infectious stage 2 to Removed (Recovered) at rate γ"/>
  </compartments>

  <compartments PrimaryName="Removed" SecondaryName="Recovered/removed (R)"
                population="0">
    <!-- Terminal in this model (no outgoing flows) -->
  </compartments>

  <!-- ===================== -->
  <!-- Notes on mapping to paper equations -->
  <!-- ===================== -->
  <!--
    Model equations provided (discrete-time, step δt):
      S(t+δt)  = S(t)  - λ(t) δt * S(t)
      E1(t+δt) = E1(t) - σ δt * E1(t) + λ(t) δt * S(t)
      E2(t+δt) = E2(t) - σ δt * E2(t) + σ δt * E1(t)
      I1(t+δt) = I1(t) - γ δt * I1(t) + σ δt * E2(t)
      I2(t+δt) = I2(t) - γ δt * I2(t) + γ δt * I1(t)
      R(t+δt)  = R(t)  + γ δt * I2(t)

    Force of infection (paper Eq. (1)):
      λ(t) = β * I(t) / N, with I(t)=I1(t)+I2(t)

    Flow typing per metamodel:
      - Infection S→E1 is a ContactFlow (depends on S and infectious compartment).
      - Progression/recovery flows are RateFlow (single-compartment proportional).
  -->

</seir:SEIRModel>
```
