<?xml version="1.0" encoding="UTF-8"?>
<seir:SEIRModel xmi:version="2.0"
    xmlns:xmi="http://www.omg.org/XMI"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xmlns:seir="http://example.com/seirmodel">

  <!-- ===================== -->
  <!-- Parameters (symbolic) -->
  <!-- ===================== -->
  <parameters name="β" type="CONSTANT"
              description="Transmission rate parameter (rate at which new infections take place given a contact) used in force of infection λ(t) ∝ β * (infectious prevalence). Paper assumes homogeneous mixing."
              unit="per day"/>

  <parameters name="σ" type="CONSTANT"
              description="Daily progression rate through latent stages: E₁→E₂ and E₂→I₁. Implies mean incubation/latent period d_L = 2*(1/σ). Treated as known from previous studies."
              unit="per day"/>

  <parameters name="γ" type="CONSTANT"
              description="Daily progression rate through infectious stages: I₁→I₂ and I₂→R. Implies mean infectious period d_I = 2*(1/γ). Treated as known from previous studies."
              unit="per day"/>

  <parameters name="λ(t)" type="EXPRESSION"
              expression="β * (I₁ + I₂)"
              description="Force of infection at time t. Text: proportional to number of infectious people and constant transmission rate β. (Population scaling not specified in extracted text; homogeneous mixing assumed.)"
              unit="per day"/>

  <parameters name="π" type="CONSTANT"
              description="Initial susceptibility (fraction of population non-immune at start). Appears in parameter set Θ_T=(π, λ₀, ψ, σ, γ)."
              unit="dimensionless"/>

  <parameters name="λ₀" type="CONSTANT"
              description="Initial infection rate / initial force of infection parameter used in early exponential growth re-parameterization (EGR)."
              unit="per day"/>

  <parameters name="ψ" type="CONSTANT"
              description="Early exponential growth rate (EGR) parameter used to re-parameterize the transmission model under initial exponential growth assumption."
              unit="per day"/>

  <parameters name="p_ICU" type="CONSTANT"
              description="Probability of ICU/HDU admission given infection (severity measure)."
              unit="dimensionless"/>

  <parameters name="μ_ICU|E" type="CONSTANT"
              description="Mean delay from infection/exposure to ICU/HDU admission; waiting time modeled as Gamma distributed. Treated as known from previous studies."
              unit="days"/>

  <parameters name="σ²_ICU|E" type="CONSTANT"
              description="Variance of delay from infection/exposure to ICU/HDU admission (Gamma waiting time). Treated as known from previous studies."
              unit="days^2"/>

  <parameters name="η" type="CONSTANT"
              description="Negative Binomial dispersion parameter for weekly ICU/HDU admission counts (observation model)."
              unit="dimensionless"/>

  <!-- ===================== -->
  <!-- Compartments (states) -->
  <!-- ===================== -->
  <compartments PrimaryName="Susceptible" SecondaryName="S(t)" population="0">
    <!-- Infection: S -> E1 via contact with infectious -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.0"
                   contactCompartment="//@compartments.3"
                   target="//@compartments.1"
                   description="Infection of susceptibles; force of infection λ(t) proportional to infectious individuals and β (homogeneous mixing)."/>
    <!-- Note: contactCompartment points to I₁; model uses I₁+I₂ in λ(t) but metamodel allows one contactCompartment.
         We represent infectious pressure via I₁ as the contact compartment; I₂ is included structurally as infectious stage below. -->
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="E₁(t)" population="0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.1"
                   target="//@compartments.2"
                   description="Latent progression stage 1: E₁ → E₂ at rate σ."/>
  </compartments>

  <compartments PrimaryName="Exposed" SecondaryName="E₂(t)" population="0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.1"
                   target="//@compartments.3"
                   description="Latent progression stage 2: E₂ → I₁ at rate σ."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="I₁(t)" population="0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.2"
                   target="//@compartments.4"
                   description="Infectious progression stage 1: I₁ → I₂ at rate γ."/>
  </compartments>

  <compartments PrimaryName="Infectious" SecondaryName="I₂(t)" population="0">
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.2"
                   target="//@compartments.5"
                   description="Recovery/removal: I₂ → R at rate γ."/>
  </compartments>

  <compartments PrimaryName="Recovered" SecondaryName="R(t)" population="0"/>

  <!-- ICU is shown in the figure as an auxiliary node ICU_t.
       In the paper, ICU admissions are generated via a convolution of past infections with a Gamma delay and probability p_ICU,
       and weekly counts are observed with NegBin(μ_w, η). This is not a standard Markovian compartment flow.
       We include it as a compartmental state placeholder to reflect the diagram, but note that its dynamics are observational. -->
  <compartments PrimaryName="Hospitalized" SecondaryName="ICU_t" population="0">
    <!-- No outgoing flows: ICU_t is an observation-linked quantity, not part of the SEIR progression chain in the paper. -->
  </compartments>

</seir:SEIRModel>
