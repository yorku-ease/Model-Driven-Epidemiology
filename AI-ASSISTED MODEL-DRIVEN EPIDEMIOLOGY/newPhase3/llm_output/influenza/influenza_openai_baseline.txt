```xml
<?xml version="1.0" encoding="UTF-8"?>
<seir:SEIRModel xmi:version="2.0"
    xmlns:xmi="http://www.omg.org/XMI"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xmlns:seir="http://example.com/seirmodel">

  <!-- ===================== -->
  <!-- Parameters (symbolic) -->
  <!-- ===================== -->
  <parameters name="N" type="VARIABLE"
              description="Total population size for the season (England). Used in force of infection λ(t)=β I(t)/N."
              unit="people"/>

  <parameters name="δt" type="CONSTANT" expression="0.25"
              description="Discrete time step length (a quarter of a day)."
              unit="day"/>

  <parameters name="β" type="CONSTANT"
              description="Transmission rate parameter in force of infection λ(t)=β I(t)/N (rate at which new infections occur given a contact)."
              unit="1/day"/>

  <parameters name="σ" type="CONSTANT" expression="1"
              description="Daily rate of progression through exposed stages (E₁→E₂ and E₂→I₁). Paper assumes known; implies mean incubation period d_L = 2/σ."
              unit="1/day"/>

  <parameters name="γ" type="CONSTANT" expression="0.5797"
              description="Daily rate of progression through infectious stages (I₁→I₂ and I₂→R). Paper assumes known; implies mean infectious period d_I = 2/γ."
              unit="1/day"/>

  <parameters name="p_ICU" type="CONSTANT"
              description="Probability of ICU/HDU admission given infection (used in observational model; not a transmission flow in the SEIR system)."
              unit="dimensionless"/>

  <parameters name="μ_ICU|E" type="CONSTANT" expression="5.708"
              description="Mean time from infection to ICU/HDU admission (Gamma waiting time in observational model)."
              unit="day"/>

  <parameters name="σ²_ICU|E" type="CONSTANT" expression="18.24"
              description="Variance of time from infection to ICU/HDU admission (Gamma waiting time in observational model)."
              unit="day^2"/>

  <parameters name="η" type="CONSTANT"
              description="Negative binomial dispersion parameter for weekly ICU/HDU admission counts (observational model)."
              unit="dimensionless"/>

  <!-- Force of infection as given in the paper -->
  <parameters name="λ(t)" type="EXPRESSION" expression="β * I_tot / N"
              description="Force of infection at time t, assuming homogeneous mixing; I_tot is total infectious (I₁+I₂)."
              unit="1/day"/>

  <parameters name="I_tot" type="EXPRESSION" expression="I₁ + I₂"
              description="Total infectious individuals at time t."
              unit="people"/>

  <!-- ================= -->
  <!-- Compartments      -->
  <!-- ================= -->
  <compartments PrimaryName="Susceptible" SecondaryName="All"
                description="S(t): susceptible individuals."/>

  <compartments PrimaryName="Exposed" SecondaryName="Stage 1"
                description="E₁(t): first exposed (latent) stage."/>

  <compartments PrimaryName="Exposed" SecondaryName="Stage 2"
                description="E₂(t): second exposed (latent) stage."/>

  <compartments PrimaryName="Infectious" SecondaryName="Stage 1"
                description="I₁(t): first infectious stage."/>

  <compartments PrimaryName="Infectious" SecondaryName="Stage 2"
                description="I₂(t): second infectious stage."/>

  <compartments PrimaryName="Removed" SecondaryName="All"
                description="R(t): removed/recovered individuals."/>

  <!-- ================= -->
  <!-- Flows             -->
  <!-- ================= -->

  <!-- Transmission: S -> E1 with frequency-dependent contact with total infectious -->
  <compartments xsi:nil="true"/>

  <!-- NOTE: In this metamodel, ContactFlow references a single contactCompartment.
       The paper uses I(t)=I₁+I₂ in λ(t)=β I(t)/N. We implement contact with I_tot
       by pointing to Infectious Stage 1 and using λ(t) expression parameter that
       already includes I_tot; this keeps the structure faithful to the paper's equation. -->

  <!-- S -> E1 -->
  <outgoingFlows xsi:type="seir:ContactFlow"
                 contactRateParameter="//@parameters.9"
                 contactCompartment="//@compartments.3"
                 target="//@compartments.1"
                 description="Infection: Susceptible → Exposed stage 1 at rate λ(t)·S(t), where λ(t)=β·(I₁+I₂)/N."/>

  <!-- Progression: E1 -> E2 -->
  <outgoingFlows xsi:type="seir:RateFlow"
                 rateParameter="//@parameters.4"
                 target="//@compartments.2"
                 description="Latent progression: Exposed stage 1 → Exposed stage 2 at rate σ."/>

  <!-- Progression: E2 -> I1 -->
  <outgoingFlows xsi:type="seir:RateFlow"
                 rateParameter="//@parameters.4"
                 target="//@compartments.3"
                 description="Becoming infectious: Exposed stage 2 → Infectious stage 1 at rate σ."/>

  <!-- Progression: I1 -> I2 -->
  <outgoingFlows xsi:type="seir:RateFlow"
                 rateParameter="//@parameters.5"
                 target="//@compartments.4"
                 description="Infectious progression: Infectious stage 1 → Infectious stage 2 at rate γ."/>

  <!-- Recovery/Removal: I2 -> R -->
  <outgoingFlows xsi:type="seir:RateFlow"
                 rateParameter="//@parameters.5"
                 target="//@compartments.5"
                 description="Removal/recovery: Infectious stage 2 → Removed at rate γ."/>

  <!-- ========================= -->
  <!-- Initial populations       -->
  <!-- ========================= -->
  <!-- The paper defines initial state via (π, λ0, ψ) etc., but does not give explicit
       numeric initial compartment counts in the main text. Therefore populations are
       left unspecified here (to be provided at simulation/estimation time). -->

</seir:SEIRModel>
```
