<?xml version="1.0" encoding="UTF-8"?>
<seir:SEIRModel xmi:version="2.0"
    xmlns:xmi="http://www.omg.org/XMI"
    xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
    xmlns:seir="http://example.com/seirmodel">

  <!-- ===================== -->
  <!-- Parameters (symbolic) -->
  <!-- ===================== -->
  <parameters name="β" type="CONSTANT"
              description="Transmission rate parameter (per-contact transmission scaled by contact rate). Used in force of infection λ(t) which is proportional to β and the number of infectious individuals."
              unit="1/time"/>
  <parameters name="λ(t)" type="EXPRESSION"
              description="Force of infection (rate at which a susceptible becomes infected). Assumed proportional to the number of infectious people at time t and transmission rate β (homogeneous mixing)."
              unit="1/time"/>
  <parameters name="σ" type="CONSTANT"
              description="Daily progression rate through latent (exposed) stages: E₁→E₂ and E₂→I₁. Implies mean incubation period d_L = 2*(1/σ) days."
              unit="1/day"/>
  <parameters name="γ" type="CONSTANT"
              description="Daily progression/recovery rate through infectious stages: I₁→I₂ and I₂→R. Implies mean infectious period d_I = 2*(1/γ) days."
              unit="1/day"/>
  <parameters name="δt" type="VARIABLE"
              description="Discrete time step used in the difference-equation form of the compartmental model."
              unit="day"/>

  <!-- ===================== -->
  <!-- Compartments (states) -->
  <!-- One per equation       -->
  <!-- ===================== -->

  <!-- 0 -->
  <compartments PrimaryName="Susceptible" population="0">
    <!-- S -> E1 : infection at rate λ(t) (implemented as ContactFlow with contactRateParameter=λ(t)) -->
    <outgoingFlows xsi:type="seir:ContactFlow"
                   contactRateParameter="//@parameters.1"
                   contactCompartment="//@compartments.4"
                   target="//@compartments.1"
                   description="Infection of susceptible individuals; force of infection λ(t) (proportional to β and infectious individuals under homogeneous mixing)."/>
  </compartments>

  <!-- 1 -->
  <compartments PrimaryName="Exposed" SecondaryName="Stage 1 (E1)" population="0">
    <!-- E1 -> E2 : progression at rate σ -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.2"
                   target="//@compartments.2"
                   description="Progression through latent/incubation stage: E1 → E2 at rate σ."/>
  </compartments>

  <!-- 2 -->
  <compartments PrimaryName="Exposed" SecondaryName="Stage 2 (E2)" population="0">
    <!-- E2 -> I1 : progression at rate σ -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.2"
                   target="//@compartments.3"
                   description="Progression from latent to infectious: E2 → I1 at rate σ."/>
  </compartments>

  <!-- 3 -->
  <compartments PrimaryName="Infectious" SecondaryName="Stage 1 (I1)" population="0">
    <!-- I1 -> I2 : progression at rate γ -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.3"
                   target="//@compartments.4"
                   description="Progression through infectious stage: I1 → I2 at rate γ."/>
  </compartments>

  <!-- 4 -->
  <compartments PrimaryName="Infectious" SecondaryName="Stage 2 (I2)" population="0">
    <!-- I2 -> R : recovery/removal at rate γ -->
    <outgoingFlows xsi:type="seir:RateFlow"
                   rateParameter="//@parameters.3"
                   target="//@compartments.5"
                   description="Recovery/removal: I2 → R at rate γ."/>
  </compartments>

  <!-- 5 -->
  <compartments PrimaryName="Recovered" population="0"/>
</seir:SEIRModel>
